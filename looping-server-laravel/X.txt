  /**
 * Store a newly created order (POST /api/infos) WITH STOCK CHECK
 */
public function store(Request $request, StockService $stockService): JsonResponse
{
    try {
        Log::info('=== ORDER STORE START ===');

        // Validation
        $validated = $request->validate([
            'customer_name' => 'required|string|max:255',
            'customer_email' => 'required|email',
            'customer_phone' => 'required|string',
            'customer_cpf' => 'required|string',
            'total' => 'required|numeric|min:0',
            'subtotal' => 'required|numeric|min:0',
            'shipping' => 'required|numeric|min:0',
            'discount' => 'sometimes|numeric|min:0',
            'payment_method' => 'required|string|in:pix,credit,debit,boleto',
            'payment_status' => 'required|string|in:pending,paid,failed,refunded',
            'status' => 'required|string|in:pending,processing,shipped,delivered,cancelled',
            'shipping_street' => 'required|string',
            'shipping_number' => 'required|string',
            'shipping_complement' => 'nullable|string',
            'shipping_neighborhood' => 'required|string',
            'shipping_city' => 'required|string',
            'shipping_state' => 'required|string|size:2',
            'shipping_zip_code' => 'required|string',
            'items' => 'required|array|min:1',
            'items.*.product_name' => 'required|string',
            'items.*.quantity' => 'required|integer|min:1',
            'items.*.price' => 'required|numeric|min:0',
            'items.*.total' => 'required|numeric|min:0',
            'items.*.image' => 'sometimes|string|nullable',
            'items.*.color' => 'sometimes|string|nullable',
            'items.*.size' => 'sometimes|string|nullable',
        ]);

        // ✅ ENSURE PRODUCT NAME IS PRESENT IN EACH ITEM
        $itemsWithProductName = [];
        foreach ($validated['items'] as $item) {
            if (empty($item['product_name'])) {
                return response()->json([
                    'success' => false,
                    'message' => 'Product name is required for all items'
                ], 400);
            }
            
            $itemsWithProductName[] = [
                'product_name' => $item['product_name'],
                'quantity' => $item['quantity'],
                'price' => $item['price'],
                'total' => $item['total'],
                'image' => $item['image'] ?? null,
                'color' => $item['color'] ?? null,
                'size' => $item['size'] ?? null,
            ];
        }

        // ✅ OPTIONAL: Check stock before creating order
        if ($request->has('check_stock') && $request->check_stock == true) {
            $stockCheck = $stockService->checkStockAvailability($itemsWithProductName);
            $unavailableItems = array_filter($stockCheck, function($result) {
                return $result['available'] === false;
            });
            
            if (!empty($unavailableItems)) {
                return response()->json([
                    'success' => false,
                    'message' => 'Some items are out of stock',
                    'stock_check' => $stockCheck,
                    'unavailable_items' => $unavailableItems
                ], 400);
            }
        }

        // Generate order number
        $orderNumber = $request->order_number ?? 'ORD-' . date('Ymd') . '-' . strtoupper(substr(uniqid(), -6));

        // Prepare the data
        $infoData = [
            'order_number' => $orderNumber,
            'customer_name' => $validated['customer_name'],
            'customer_email' => $validated['customer_email'],
            'customer_phone' => $validated['customer_phone'],
            'customer_cpf' => $validated['customer_cpf'],
            'total' => $validated['total'],
            'subtotal' => $validated['subtotal'],
            'shipping' => $validated['shipping'],
            'discount' => $validated['discount'] ?? 0,
            'payment_method' => $validated['payment_method'],
            'payment_status' => $validated['payment_status'],
            'stock_updated' => false,
            'stock_updated_at' => null,
            'status' => $validated['status'],
            'shipping_street' => $validated['shipping_street'],
            'shipping_number' => $validated['shipping_number'],
            'shipping_complement' => $validated['shipping_complement'] ?? null,
            'shipping_neighborhood' => $validated['shipping_neighborhood'],
            'shipping_city' => $validated['shipping_city'],
            'shipping_state' => $validated['shipping_state'],
            'shipping_zip_code' => $validated['shipping_zip_code'],
            'items' => json_encode($itemsWithProductName),
            'metadata' => $request->metadata ? json_encode($request->metadata) : null,
        ];

        Log::info('Creating order with items:', [
            'order_number' => $orderNumber,
            'items' => $itemsWithProductName
        ]);

        $info = Info::create($infoData);

        // ✅ If payment is already marked as paid, deduct stock immediately
        if ($validated['payment_status'] === 'paid') {
            $stockUpdated = $stockService->deductStockFromOrder($info);
            if (!$stockUpdated) {
                $info->delete();
                return response()->json([
                    'success' => false,
                    'message' => 'Failed to create order. Insufficient stock available.',
                    'order_number' => $orderNumber
                ], 400);
            }
        }

        return response()->json([
            'success' => true,
            'message' => 'Order created successfully',
            'data' => [
                'order_id' => $info->id,
                'order_number' => $info->order_number,
                'items' => $info->parsed_items,
                'stock_updated' => $info->stock_updated
            ]
        ], 201);

    } catch (\Illuminate\Validation\ValidationException $e) {
        Log::error('=== ORDER STORE VALIDATION ERROR ===');
        Log::error('Validation errors:', $e->errors());

        return response()->json([
            'success' => false,
            'message' => 'Validation failed',
            'errors' => $e->errors(),
        ], 422);

    } catch (Exception $e) {
        Log::error('=== ORDER STORE ERROR ===');
        Log::error('Error: ' . $e->getMessage());

        return response()->json([
            'success' => false,
            'message' => 'Failed to create order',
            'error' => $e->getMessage(),
        ], 500);
    }
}

    /**
     * ✅ Confirm payment and update stock
     * POST /api/infos/{id}/confirm-payment
     */
    public function confirmPayment($id, Request $request, StockService $stockService): JsonResponse
    {
        try {
            Log::info('=== CONFIRM PAYMENT START ===', ['order_id' => $id]);

            // Find the order
            $info = Info::find($id);
            if (!$info) {
                return response()->json([
                    'success' => false,
                    'message' => 'Order not found'
                ], 404);
            }

            // Check if payment is already confirmed
            if ($info->payment_status === 'paid' && $info->stock_updated) {
                return response()->json([
                    'success' => true,
                    'message' => 'Payment already confirmed and stock updated',
                    'data' => [
                        'order_id' => $info->id,
                        'order_number' => $info->order_number,
                        'payment_status' => $info->payment_status,
                        'stock_updated' => $info->stock_updated,
                        'stock_updated_at' => $info->stock_updated_at
                    ]
                ]);
            }

            // Update payment status
            $info->payment_status = 'paid';
            $info->status = 'processing';
            
            // Deduct stock
            $stockUpdated = $stockService->deductStockFromOrder($info);
            
            if (!$stockUpdated) {
                return response()->json([
                    'success' => false,
                    'message' => 'Failed to update stock. Please check stock availability.'
                ], 400);
            }

            $info->save();

            Log::info('Payment confirmed and stock updated', [
                'order_id' => $info->id,
                'order_number' => $info->order_number,
                'stock_updated' => $info->stock_updated,
                'stock_updated_at' => $info->stock_updated_at
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Payment confirmed and stock updated successfully',
                'data' => [
                    'order_id' => $info->id,
                    'order_number' => $info->order_number,
                    'payment_status' => $info->payment_status,
                    'status' => $info->status,
                    'stock_updated' => $info->stock_updated,
                    'stock_updated_at' => $info->stock_updated_at
                ]
            ]);

        } catch (Exception $e) {
            Log::error('Confirm payment error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Failed to confirm payment',
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * ✅ Check stock availability before placing order
     * POST /api/infos/check-stock
     */
    public function checkStock(Request $request, StockService $stockService): JsonResponse
    {
        try {
            $validated = $request->validate([
                'items' => 'required|array|min:1',
                'items.*.product_name' => 'required|string',
                'items.*.quantity' => 'required|integer|min:1',
                'items.*.color' => 'sometimes|string|nullable',
                'items.*.size' => 'sometimes|string|nullable',
            ]);

            $stockCheck = $stockService->checkStockAvailability($validated['items']);
            
            $allAvailable = true;
            $messages = [];
            $unavailableItems = [];
            
            foreach ($stockCheck as $key => $result) {
                if (!$result['available']) {
                    $allAvailable = false;
                    $item = $validated['items'][$key];
                    $messages[] = "Estoque insuficiente para {$item['product_name']}: Disponível {$result['available_stock']}, Solicitado {$item['quantity']}";
                    $unavailableItems[] = [
                        'product_name' => $item['product_name'],
                        'requested_quantity' => $item['quantity'],
                        'available_stock' => $result['available_stock'],
                        'color' => $item['color'] ?? null,
                        'size' => $item['size'] ?? null
                    ];
                }
            }

            return response()->json([
                'success' => true,
                'all_available' => $allAvailable,
                'stock_check' => $stockCheck,
                'messages' => $messages,
                'unavailable_items' => $unavailableItems,
                'debug' => [
                    'items_count' => count($validated['items']),
                    'timestamp' => now()->format('c')
                ]
            ]);

        } catch (Exception $e) {
            Log::error('Check stock error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Failed to check stock',
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * ✅ Manual stock sync for an order
     * POST /api/infos/{id}/sync-stock
     */
    public function syncStock($id, StockService $stockService): JsonResponse
    {
        try {
            Log::info('=== MANUAL STOCK SYNC START ===', ['order_id' => $id]);

            $info = Info::find($id);
            if (!$info) {
                return response()->json([
                    'success' => false,
                    'message' => 'Order not found'
                ], 404);
            }

            // Check if order is eligible for stock deduction
            if (!$info->canDeductStock()) {
                return response()->json([
                    'success' => false,
                    'message' => 'Order is not eligible for stock deduction',
                    'details' => [
                        'payment_status' => $info->payment_status,
                        'status' => $info->status,
                        'stock_updated' => $info->stock_updated
                    ]
                ], 400);
            }

            $stockUpdated = $stockService->deductStockFromOrder($info);
            
            if (!$stockUpdated) {
                return response()->json([
                    'success' => false,
                    'message' => 'Failed to sync stock'
                ], 400);
            }

            return response()->json([
                'success' => true,
                'message' => 'Stock synced successfully',
                'data' => [
                    'order_id' => $info->id,
                    'order_number' => $info->order_number,
                    'stock_updated' => $info->stock_updated,
                    'stock_updated_at' => $info->stock_updated_at
                ]
            ]);

        } catch (Exception $e) {
            Log::error('Manual stock sync error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Failed to sync stock',
                'error' => $e->getMessage()
            ], 500);
        }
    }

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\Log;

class Info extends Model
{
    use HasFactory;

    protected $table = 'orders';

    protected $fillable = [
        'order_number',
        'customer_name',
        'customer_email',
        'customer_phone',
        'customer_cpf',
        'total',
        'subtotal',
        'shipping',
        'discount',
        'payment_method',
        'payment_status',
        'stock_updated',           // ✅ TRACKS IF STOCK WAS DEDUCTED
        'stock_updated_at',        // ✅ WHEN STOCK WAS DEDUCTED
        'status',
        'shipping_street',
        'shipping_number',
        'shipping_complement',
        'shipping_neighborhood',
        'shipping_city',
        'shipping_state',
        'shipping_zip_code',
        'items',                   // ✅ CONTAINS ORDERED QUANTITIES
        'metadata'
    ];

    protected $casts = [
        'total' => 'decimal:2',
        'subtotal' => 'decimal:2',
        'shipping' => 'decimal:2',
        'discount' => 'decimal:2',
        'stock_updated' => 'boolean',  // ✅ Cast as boolean
        'items' => 'array',            // ✅ Cast as array (contains quantities)
        'metadata' => 'array',
        'created_at' => 'datetime',
        'updated_at' => 'datetime',
        'stock_updated_at' => 'datetime',  // ✅ Cast as datetime
    ];

    /**
     * Boot method
     */
    protected static function boot()
    {
        parent::boot();

        static::retrieved(function ($order) {
            try {
                if ($order->items && is_string($order->items)) {
                    $order->items = json_decode($order->items, true);
                }
            
                if ($order->metadata && is_string($order->metadata)) {
                    $order->metadata = json_decode($order->metadata, true);
                }
            } catch (\Exception $e) {
                Log::error('Error decoding JSON fields for order ' . $order->id . ': ' . $e->getMessage());
            }
        });
    }

    /**
     * ✅ Check if stock needs to be updated
     */
    public function needsStockUpdate(): bool
    {
        return $this->payment_status === 'paid' && !$this->stock_updated;
    }

    /**
     * ✅ Mark stock as updated
     */
    public function markStockAsUpdated(): void
    {
        $this->stock_updated = true;
        $this->stock_updated_at = now();
        $this->save();
    }

    /**
     * ✅ Check if order is eligible for stock deduction
     */
    public function canDeductStock(): bool
    {
        return in_array($this->payment_status, ['paid']) 
            && !in_array($this->status, ['cancelled', 'refunded'])
            && !$this->stock_updated;
    }

    /**
     * ✅ Check if order should restore stock
     */
    public function shouldRestoreStock(): bool
    {
        return ($this->payment_status === 'refunded' || $this->status === 'cancelled')
            && $this->stock_updated;
    }

    /**
     * ✅ Scope for orders that need stock update
     */
    public function scopeNeedsStockUpdate($query)
    {
        return $query->where('payment_status', 'paid')
                     ->where('stock_updated', false);
    }

    /**
     * ✅ Scope for paid orders
     */
    public function scopePaid($query)
    {
        return $query->where('payment_status', 'paid');
    }

    /**
     * ✅ Get parsed items with quantities
     */
    public function getParsedItemsAttribute(): array
    {
        if (is_array($this->items)) {
            return $this->items;
        }

        if (is_string($this->items)) {
            return json_decode($this->items, true) ?? [];
        }

        return [];
    }

    /**
     * ✅ Get total quantity of items in order
     */
    public function getTotalQuantityAttribute(): int
    {
        $total = 0;
        foreach ($this->parsed_items as $item) {
            $total += $item['quantity'] ?? 0;
        }
        return $total;
    }
}

                'error' => $e->getMessage()
            ], 500);
        }
    }

    public function destroy($id): JsonResponse
    {
        try {
            Log::order("Attempting to delete stock item with ID: {$id}");

            $stock = Stock::find($id);
            
            if (!$stock) {
                Log::warning("Stock item not found with ID: {$id}");
                return response()->json([
                    'success' => false,
                    'message' => 'Item de estoque não encontrado'
                ], 404);
            }

            $productInfo = [
                'id' => $stock->id,
                'product_id' => $stock->product_id,
                'name' => $stock->name,
                'color' => $stock->color,
                'size' => $stock->size
            ];

            $stock->delete();

            Log::order("Stock item deleted successfully", $productInfo);

            return response()->json([
                'success' => true,
                'message' => 'Variante excluída com sucesso',
                'deleted_item' => $productInfo
            ]);

        } catch (\Exception $e) {
            Log::error('Delete stock item error: ' . $e->getMessage(), [
                'stock_id' => $id,
                'exception' => $e->getTraceAsString()
            ]);
            
            return response()->json([
                'success' => false,
                'message' => 'Erro ao excluir variante: ' . $e->getMessage()
            ], 500);
        }
    }

    public function destroyByProductId($productId): JsonResponse
    {
        try {
            Log::order("Attempting to delete all stock items for product_id: {$productId}");

            $existingItems = Stock::where('product_id', $productId)->get();
            
            if ($existingItems->isEmpty()) {
                Log::warning("No stock items found for product_id: {$productId}");
                return response()->json([
                    'success' => false,
                    'message' => 'Nenhum produto encontrado com este ID'
                ], 404);
            }

            $itemsInfo = $existingItems->map(function ($item) {
                return [
                    'id' => $item->id,
                    'product_id' => $item->product_id,
                    'name' => $item->name,
                    'color' => $item->color,
                    'size' => $item->size
                ];
            })->toArray();

            $deletedCount = Stock::where('product_id', $productId)->delete();

            Log::order("Product and all variants deleted successfully", [
                'product_id' => $productId,
                'deleted_count' => $deletedCount,
                'deleted_items' => $itemsInfo
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Produto e todas as variantes excluídos com sucesso',
                'deleted_count' => $deletedCount,
                'deleted_items' => $itemsInfo
            ]);

        } catch (\Exception $e) {
            Log::error('Delete product error: ' . $e->getMessage(), [
                'product_id' => $productId,
                'exception' => $e->getTraceAsString()
            ]);
            
            return response()->json([
                'success' => false,
                'message' => 'Erro ao excluir produto: ' . $e->getMessage()
            ], 500);
        }

    public $timestamps = false;
    
    protected $fillable = [
        'product_id', 
        'name', 
        'sku', 
        'category', 
        'supplier',

        'stock' => 'integer',      // ✅ Cast as integer
        'last_updated' => 'date'
    ];

    /**
     * Get grouped stock by product
     */
    public static function getGroupedStock()
    {
        if (!Schema::hasTable('stock')) {
            return collect();
        }

        return self::orderBy('product_id')
            ->orderBy('color')
            ->orderBy('size')
            ->get()
            ->groupBy('product_id')
            ->map(function ($items, $productId) {
                $firstItem = $items->first();
                
                return [
                    'id' => $firstItem->id,
                    'product_id' => $productId,
                    'name' => $firstItem->name,
                    'category' => $firstItem->category,
                    'supplier' => $firstItem->supplier,
                    'cost' => (float) $firstItem->cost,
                    'profit' => (float) $firstItem->profit,
                    'sale_price' => (float) $firstItem->sale_price,
                    'total_value' => (float) $firstItem->total_value,
                    'last_updated' => $firstItem->last_updated,
                    // Removed created_at and updated_at references
                    'variants' => $items->map(function ($item) {
                        return [
                            'id' => $item->id,
                            'color' => $item->color,
                            'size' => $item->size,
                            'stock' => $item->stock, // ✅ AVAILABLE QUANTITY
                            'sku' => $item->sku,
                        ];
                    })->values()->toArray()
                ];
            })->values();
    }

    public static function getLowStock()
    {
        if (!Schema::hasTable('stock')) {
            return collect();
        }

        return self::where('stock', '>', 0)
            ->where('stock', '<=', 5)
            ->get();
    }

    public static function getOutOfStock()
    {
        if (!Schema::hasTable('stock')) {
            return collect();
        }

        return self::where('stock', 0)->get();
    }

<?php

namespace App\Services;

use App\Models\Stock;
use App\Models\Info;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class StockService
{
    /**
     * ✅ Deduct stock for a paid order
     */
    public function deductStockFromOrder(Info $order): bool
    {
        try {
            Log::info('=== DEDUCT STOCK FROM ORDER START ===', [
                'order_id' => $order->id,
                'order_number' => $order->order_number,
                'payment_status' => $order->payment_status,
                'stock_updated_before' => $order->stock_updated
            ]);

            // If stock already updated, return true
            if ($order->stock_updated) {
                Log::info('Stock already deducted for order');
                return true;
            }

            // Only deduct stock for paid orders
            if ($order->payment_status !== 'paid') {
                Log::warning('Cannot deduct stock for unpaid order');
                return false;
            }

            $items = $order->parsed_items;
            if (empty($items)) {
                Log::warning('No items in order');
                return false;
            }

            DB::beginTransaction();

            // Check all items have enough stock first
            foreach ($items as $item) {
                if (!$this->checkItemStockAvailability($item)) {
                    throw new \Exception("Insufficient stock for: " . ($item['product_name'] ?? 'Unknown'));
                }
            }

            // Deduct stock for each item
            foreach ($items as $item) {
                $this->deductItemStock($item);
            }

            // Mark order as having stock deducted
            $order->stock_updated = true;
            $order->stock_updated_at = now();
            $order->save();

            DB::commit();

            Log::info('=== STOCK DEDUCTION COMPLETE ===', [
                'order_id' => $order->id,
                'items_count' => count($items),
                'stock_updated' => $order->stock_updated,
                'stock_updated_at' => $order->stock_updated_at
            ]);

            return true;

        } catch (\Exception $e) {
            DB::rollBack();
            Log::error('=== STOCK DEDUCTION FAILED ===', [
                'order_id' => $order->id,
                'error' => $e->getMessage(),
                'trace' => $e->getTraceAsString()
            ]);
            return false;
        }
    }

    /**
     * ✅ Check if item has enough stock
     */
    private function checkItemStockAvailability(array $item): bool
    {
        $productName = $item['product_name'] ?? null;
        $quantity = $item['quantity'] ?? 0;
        $color = $item['color'] ?? null;
        $size = $item['size'] ?? null;

        if (!$productName || $quantity <= 0) {
            return false;
        }

        $stock = $this->findStockItem($productName, $color, $size);
        
        if (!$stock) {
            Log::warning('Stock item not found', [
                'product_name' => $productName,
                'color' => $color,
                'size' => $size
            ]);
            return false;
        }

        $available = $stock->stock >= $quantity;
        
        if (!$available) {
            Log::warning('Insufficient stock', [
                'product_name' => $productName,
                'required' => $quantity,
                'available' => $stock->stock
            ]);
        }

        return $available;
    }

    /**
     * ✅ Deduct stock for a single item
     */
    private function deductItemStock(array $item): void
    {
        $productName = $item['product_name'] ?? null;
        $quantity = $item['quantity'] ?? 0;
        $color = $item['color'] ?? null;
        $size = $item['size'] ?? null;

        if (!$productName || $quantity <= 0) {
            return;
        }

        $stock = $this->findStockItem($productName, $color, $size);
        
        if (!$stock) {
            throw new \Exception("Stock not found for product: {$productName}");
        }

        // Deduct from available stock
        $stock->decrement('stock', $quantity);
        $stock->last_updated = now()->format('Y-m-d');
        $stock->save();

        Log::info('Stock deducted for item', [
            'stock_id' => $stock->id,
            'product_name' => $productName,
            'quantity' => $quantity,
            'remaining_stock' => $stock->stock,
            'color' => $color,
            'size' => $size
        ]);
    }
    
    /**
     * ✅ Restore stock for cancelled/refunded orders
     */
    public function restoreStockForOrder(Info $order): bool
    {
        try {
            Log::info('=== RESTORE STOCK FOR ORDER START ===', [
                'order_id' => $order->id,
                'order_number' => $order->order_number,
                'payment_status' => $order->payment_status,
                'stock_updated_before' => $order->stock_updated
            ]);

            // Only restore if stock was previously deducted
            if (!$order->stock_updated) {
                Log::info('Stock was not deducted for this order, nothing to restore');
                return true;
            }

            $items = $order->parsed_items;
            if (empty($items)) {
                return false;
            }

            DB::beginTransaction();

            // Restore stock for each item
            foreach ($items as $item) {
                $this->restoreItemStock($item);
            }

            // Mark order as having stock restored
            $order->stock_updated = false;
            $order->stock_updated_at = null;
            $order->save();

            DB::commit();

            Log::info('=== STOCK RESTORATION COMPLETE ===', [
                'order_id' => $order->id,
                'items_count' => count($items),
                'stock_updated' => $order->stock_updated
            ]);

            return true;

        } catch (\Exception $e) {
            DB::rollBack();
            Log::error('=== STOCK RESTORATION FAILED ===', [
                'order_id' => $order->id,
                'error' => $e->getMessage()
            ]);
            return false;
        }
    }

    /**
     * ✅ Restore stock for a single item
     */
    private function restoreItemStock(array $item): void
    {
        $productName = $item['product_name'] ?? null;
        $quantity = $item['quantity'] ?? 0;
        $color = $item['color'] ?? null;
        $size = $item['size'] ?? null;

        if (!$productName || $quantity <= 0) {
            return;
        }

        $stock = $this->findStockItem($productName, $color, $size);
        
        if (!$stock) {
            Log::warning('Stock item not found when restoring', [
                'product_name' => $productName,
                'color' => $color,
                'size' => $size
            ]);
            return;
        }

        // Restore to available stock
        $stock->increment('stock', $quantity);
        $stock->last_updated = now()->format('Y-m-d');
        $stock->save();

        Log::info('Stock restored for item', [
            'stock_id' => $stock->id,
            'product_name' => $productName,
            'quantity' => $quantity,
            'new_stock' => $stock->stock
        ]);
    }

    /**
     * ✅ Find stock item by name, color, and size
     */
    private function findStockItem(string $productName, ?string $color, ?string $size): ?Stock
    {
        $stock = Stock::where('name', $productName)
            ->when($color && $color !== 'Não especificada', function($query) use ($color) {
                return $query->where('color', $color);
            })
            ->when($size && $size !== 'Não especificado', function($query) use ($size) {
                return $query->where('size', $size);
            })
            ->first();

        // Fallback: Try without color/size
        if (!$stock) {
            $stock = Stock::where('name', $productName)->first();
        }

        return $stock;
    }

    /**
     * ✅ Check stock availability for multiple items
     */
    public function checkStockAvailability(array $items): array
    {
        $results = [];

        foreach ($items as $key => $item) {
            $productName = $item['product_name'] ?? null;
            $quantity = $item['quantity'] ?? 0;
            $color = $item['color'] ?? null;
            $size = $item['size'] ?? null;

            $result = [
                'product_name' => $productName,
                'requested_quantity' => $quantity,
                'color' => $color,
                'size' => $size,
                'available' => false,
                'available_stock' => 0,
                'message' => ''
            ];

            if (!$productName || $quantity <= 0) {
                $result['message'] = 'Invalid product data';
                $results[$key] = $result;
                continue;
            }

            $stock = $this->findStockItem($productName, $color, $size);

            if (!$stock) {
                $result['message'] = 'Product not found in stock';
            } else {
                $result['available_stock'] = $stock->stock;
                $result['available'] = $stock->stock >= $quantity;
                $result['message'] = $result['available'] 
                    ? 'Stock available' 
                    : "Insufficient stock. Available: {$stock->stock}, Requested: {$quantity}";
                $result['stock_id'] = $stock->id;
            }

            $results[$key] = $result;
        }

        return $results;
    }

    /**
     * ✅ Get all products with low stock
     */
    public function getLowStockProducts(int $threshold = 5): array
    {
        return Stock::where('stock', '>', 0)
            ->where('stock', '<=', $threshold)
            ->get()
            ->map(function ($stock) {
                return [
                    'id' => $stock->id,
                    'name' => $stock->name,
                    'color' => $stock->color,
                    'size' => $stock->size,
                    'stock' => $stock->stock,
                    'sale_price' => $stock->sale_price
                ];
            })
            ->toArray();
    }
