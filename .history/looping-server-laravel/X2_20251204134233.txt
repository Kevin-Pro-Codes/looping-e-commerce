namespace App\Http\Controllers;

use App\Models\Order;

use Illuminate\Support\Facades\Log;
use Exception;

class PixPaymentController extends Controller
{
    /**
     * Generate PIX payment for an order
     * POST /api/pix/generate
     */
    public function generatePix(Request $request): JsonResponse
    {
        DB::beginTransaction();

        try {
            Log::info('=== PIX GENERATE START ===');
            Log::info('PIX request data:', $request->all());

            // Validate input - use 'infos' table instead of 'orders'
            $validated = $request->validate([
                'order_id' => 'required|integer|exists:infos,id',
                'amount' => 'required|numeric|min:0.01',
                'payer_name' => 'required|string|max:150',
                'payer_cpf' => 'required|string|max:14',
            ]);

            // Get the order from infos table
            $order = Order::findOrFail($validated['order_id']);
            
            // Check if PIX details already exist
            $existingPix = PixDetails::where('order_id', $order->id)->first();
            if ($existingPix) {
                return response()->json([
                    'success' => true,
                    'message' => 'PIX already generated for this order',
                    'data' => $this->formatPixResponse($existingPix, $order)
                ]);
            }

            // Generate PIX data
            $pixData = $this->generatePixData($validated);
            
            // Create PIX details record - ONLY include fields that exist in your table
            $pixDetails = PixDetails::create([
                'order_id' => $order->id,
                'payer_name' => $validated['payer_name'],
                'payer_cpf' => $validated['payer_cpf'],
                'pix_key' => $pixData['pix_key'],
                'qr_code' => $pixData['qr_code'],
                'transaction_id' => $pixData['transaction_id'],
                'amount' => (float) $validated['amount'], // Ensure it's a float
                'status' => 'pending',
                // NO expires_at field - your table doesn't have it
            ]);

            // Update order payment status in infos table
            $order->update([
                'payment_status' => 'pending',
                'status' => 'awaiting_payment'
            ]);

            DB::commit();

            Log::info('PIX generated successfully', [
                'order_id' => $order->id,
                'transaction_id' => $pixData['transaction_id']
            ]);

            return response()->json([
                'success' => true,
                'message' => 'PIX generated successfully',
                'data' => $this->formatPixResponse($pixDetails, $order)
            ]);

        } catch (Exception $e) {
            DB::rollBack();
            
            Log::error('=== PIX GENERATE ERROR ===');
            Log::error('Error: ' . $e->getMessage());
            Log::error('File: ' . $e->getFile() . ':' . $e->getLine());
            Log::error('Trace: ' . $e->getTraceAsString());

            return response()->json([
                'success' => false,
                'message' => 'Failed to generate PIX payment: ' . $e->getMessage(),
                'error' => config('app.debug') ? $e->getMessage() : 'Internal server error'
            ], 500);
        }
    }

    /**
     * Generate PIX data
     */
    private function generatePixData(array $data): array
    {
        $transactionId = 'PIX_' . uniqid() . '_' . time();
        $pixCode = $this->generatePixCode($data['amount'], $transactionId);
        
        return [
            'pix_key' => 'mock-pix-key-' . $transactionId,
            'qr_code' => "https://api.qrserver.com/v1/create-qr-code/?size=300x300&data=" . urlencode($pixCode),
            'pix_code' => $pixCode,
            'transaction_id' => $transactionId,
        ];
    }

    /**
     * Generate PIX code in standard format
     */
    private function generatePixCode($amount, $transactionId): string
    {
        $formattedAmount = number_format((float) $amount, 2, '.', '');
        
        // Simplified PIX code structure (BR Code)
        return "00020126580014br.gov.bcb.pix0136{$transactionId}520400005303986540{$formattedAmount}5802BR5913MERCADO TESTE6008SAO PAULO62070503***6304" . strtoupper(substr(md5($transactionId), 0, 4));
    }

    /**
     * Format PIX response
     */
    private function formatPixResponse(PixDetails $pixDetails, Order $order): array
    {
        // Calculate expiration time (30 minutes from creation)
        $expiresAt = $pixDetails->created_at->addMinutes(30);
        
        return [
            'pix_qr_code' => $pixDetails->qr_code,
            'pix_code' => $pixDetails->pix_code ?? $this->extractPixCodeFromQr($pixDetails->qr_code),
            'transaction_id' => $pixDetails->transaction_id,
            'amount' => (float) $pixDetails->amount, // Ensure it's a number
            'expires_at' => $expiresAt->format('c'), // Calculate dynamically
            'order_id' => $order->id,
            'order_number' => $order->order_number,
        ];
    }

    /**
     * Extract PIX code from QR code URL if needed
     */
    private function extractPixCodeFromQr(string $qrCodeUrl): string
    {
        $parsedUrl = parse_url($qrCodeUrl);
        if (isset($parsedUrl['query'])) {
            parse_str($parsedUrl['query'], $queryParams);
            if (isset($queryParams['data'])) {
                return urldecode($queryParams['data']);
            }
        }
        
        // Fallback: generate a simple code
        return "00020126580014br.gov.bcb.pix0136mock_" . uniqid() . "520400005303986540100.005802BR5913TESTE6008SAO PAULO62070503***6304E2A9";
    }

    /**
     * Check PIX payment status
     * GET /api/pix/status/{orderId}
     */
    public function checkPaymentStatus($orderId): JsonResponse
    {
        try {
            Log::info('Checking PIX status for order:', ['order_id' => $orderId]);

            $pixDetails = PixDetails::where('order_id', $orderId)->first();
            
            if (!$pixDetails) {
                return response()->json([
                    'success' => false,
                    'message' => 'PIX details not found for this order'
                ], 404);
            }

            // Simulate payment status (in real scenario, check with PIX provider)
            $status = $this->simulatePaymentStatus($pixDetails);

            // Update status if changed
            if ($status !== $pixDetails->status) {
                $pixDetails->update(['status' => $status]);
                
                // Update order status if paid
                if ($status === 'paid') {
                    $order = Order::find($orderId);
                    if ($order) {
                        $order->update([
                            'payment_status' => 'paid',
                            'status' => 'processing'
                        ]);
                    }
                }
            }

            // Calculate expiration time dynamically
            $expiresAt = $pixDetails->created_at->addMinutes(30);

            return response()->json([
                'success' => true,
                'data' => [
                    'status' => $status,
                    'transaction_id' => $pixDetails->transaction_id,
                    'amount' => (float) $pixDetails->amount, // Ensure it's a number
                    'expires_at' => $expiresAt->format('c'),
                    'updated_at' => $pixDetails->updated_at->format('c'),
                ]
            ]);

        } catch (Exception $e) {
            Log::error('PIX Status Check Error: ' . $e->getMessage());
            
            return response()->json([
                'success' => false,
                'message' => 'Failed to check PIX status: ' . $e->getMessage()
            ], 500);
        }
    }

    /**
     * Simulate payment status for demo
     */
    private function simulatePaymentStatus(PixDetails $pixDetails): string
    {
        // Calculate expiration time (30 minutes from creation)
        $expiresAt = $pixDetails->created_at->addMinutes(30);
        
        // Auto-confirm if created more than 10 seconds ago (for demo)
        if (now()->diffInSeconds($pixDetails->created_at) > 10) {
            return 'paid';
        }
        
        // Check if expired (using calculated expiration)
        if (now()->greaterThan($expiresAt)) {
            return 'expired';
        }
        
        return $pixDetails->status;
    }

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\Log;

class Info extends Model
{
    use HasFactory;

    protected $table = 'infos';

    protected $fillable = [
        'order_number',
        'customer_name',
        'customer_email',
        'customer_phone',
        'customer_cpf',
        'total',
        'subtotal',
        'shipping',
        'discount',
        'payment_method',
        'payment_status',
        'stock_updated',           // ✅ TRACKS IF STOCK WAS DEDUCTED
        'stock_updated_at',        // ✅ WHEN STOCK WAS DEDUCTED
        'status',
        'shipping_street',
        'shipping_number',
        'shipping_complement',
        'shipping_neighborhood',
        'shipping_city',
        'shipping_state',
        'shipping_zip_code',
        'items',                   // ✅ CONTAINS ORDERED QUANTITIES
        'metadata'
    ];

    protected $casts = [
        'total' => 'decimal:2',
        'subtotal' => 'decimal:2',
        'shipping' => 'decimal:2',
        'discount' => 'decimal:2',
        'stock_updated' => 'boolean',  // ✅ Cast as boolean
        'items' => 'array',            // ✅ Cast as array (contains quantities)
        'metadata' => 'array',
        'created_at' => 'datetime',
        'updated_at' => 'datetime',
        'stock_updated_at' => 'datetime',  // ✅ Cast as datetime
    ];

    /**
     * Boot method
     */
    protected static function boot()
    {
        parent::boot();

        static::retrieved(function ($info) {
            try {
                if ($info->items && is_string($info->items)) {
                    $info->items = json_decode($info->items, true);
                }
            
                if ($info->metadata && is_string($info->metadata)) {
                    $info->metadata = json_decode($info->metadata, true);
                }
            } catch (\Exception $e) {
                Log::error('Error decoding JSON fields for info ' . $info->id . ': ' . $e->getMessage());
            }
        });
    }

    /**
     * ✅ Check if stock needs to be updated
     */
    public function needsStockUpdate(): bool
    {
        return $this->payment_status === 'paid' && !$this->stock_updated;
    }

    /**
     * ✅ Mark stock as updated
     */
    public function markStockAsUpdated(): void
    {
        $this->stock_updated = true;
        $this->stock_updated_at = now();
        $this->save();
    }

    /**
     * ✅ Check if order is eligible for stock deduction
     */
    public function canDeductStock(): bool
    {
        return in_array($this->payment_status, ['paid']) 
            && !in_array($this->status, ['cancelled', 'refunded'])
            && !$this->stock_updated;
    }

    /**
     * ✅ Check if order should restore stock
     */
    public function shouldRestoreStock(): bool
    {
        return ($this->payment_status === 'refunded' || $this->status === 'cancelled')
            && $this->stock_updated;
    }

    /**
     * ✅ Scope for orders that need stock update
     */
    public function scopeNeedsStockUpdate($query)
    {
        return $query->where('payment_status', 'paid')
                     ->where('stock_updated', false);
    }

    /**
     * ✅ Scope for paid orders
     */
    public function scopePaid($query)
    {
        return $query->where('payment_status', 'paid');
    }

    /**
     * ✅ Get parsed items with quantities
     */
    public function getParsedItemsAttribute(): array
    {
        if (is_array($this->items)) {
            return $this->items;
        }

        if (is_string($this->items)) {
            return json_decode($this->items, true) ?? [];
        }

        return [];
    }

    /**
     * ✅ Get total quantity of items in order
     */
    public function getTotalQuantityAttribute(): int
    {
        $total = 0;
        foreach ($this->parsed_items as $item) {
            $total += $item['quantity'] ?? 0;
        }
        return $total;
    }
}

    /**
     * Update info status and payment status WITH STOCK SYNC
     * PUT /api/infos/{id}/status
     */
    public function updateStatus($id, Request $request, StockService $stockService): JsonResponse
    {
        try {
            Log::info('=== INFO STATUS UPDATE START ===', ['info_id' => $id]);

            // Validate input
            $validated = $request->validate([
                'status' => 'required|string|in:pending,processing,shipped,delivered,cancelled',
                'payment_status' => 'required|string|in:pending,paid,failed,refunded'
            ]);

            // Find info
            $info = Info::find($id);
            if (!$info) {
                return response()->json([
                    'success' => false,
                    'message' => 'Info not found',
                    'debug' => ['info_id' => $id]
                ], 404);
            }

            $oldPaymentStatus = $info->payment_status;
            $oldStatus = $info->status;

            Log::info('Status change detected', [
                'old_payment_status' => $oldPaymentStatus,
                'new_payment_status' => $validated['payment_status'],
                'old_status' => $oldStatus,
                'new_status' => $validated['status']
            ]);

            // ✅ HANDLE STOCK DEDUCTION FOR NEW PAID ORDERS
            if ($oldPaymentStatus !== 'paid' && $validated['payment_status'] === 'paid') {
                Log::info('Payment changed to PAID - deducting stock');
                
                $stockUpdated = $stockService->deductStockFromOrder($info);
                
                if (!$stockUpdated) {
                    return response()->json([
                        'success' => false,
                        'message' => 'Failed to update stock. Insufficient stock available.',
                        'debug' => [
                            'info_id' => $info->id,
                            'order_number' => $info->order_number
                        ]
                    ], 400);
                }
                
                Log::info('Stock deducted successfully');
            }
            
            // ✅ HANDLE STOCK RESTORATION FOR CANCELLATIONS/REFUNDS
            if (($oldPaymentStatus === 'paid' && $validated['payment_status'] === 'refunded') ||
                ($oldStatus !== 'cancelled' && $validated['status'] === 'cancelled')) {
                Log::info('Order refunded/cancelled - restoring stock');
                
                $stockService->restoreStockForOrder($info);
            }

            // Update fields
            $info->status = $validated['status'];
            $info->payment_status = $validated['payment_status'];
            $info->save();

            Log::info('Info status updated successfully', [
                'info_id' => $info->id,
                'status' => $info->status,
                'payment_status' => $info->payment_status,
                'stock_updated' => $info->stock_updated,
                'stock_updated_at' => $info->stock_updated_at
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Info status updated successfully',
                'data' => [
                    'info_id' => $info->id,
                    'order_number' => $info->order_number,
                    'status' => $info->status,
                    'payment_status' => $info->payment_status,
                    'stock_updated' => $info->stock_updated,
                    'stock_updated_at' => $info->stock_updated_at
                ],
                'debug' => [
                    'timestamp' => now()->format('c')
                ]
            ]);

        } catch (Exception $e) {
            Log::error('=== INFO STATUS UPDATE ERROR ===');
            Log::error('Error: ' . $e->getMessage());
            Log::error('File: ' . $e->getFile() . ':' . $e->getLine());
            Log::error('Request data: ', $request->all());

            return response()->json([
                'success' => false,
                'message' => 'Failed to update info status',
                'error' => $e->getMessage(),
                'debug' => [
                    'exception' => get_class($e),
                    'message' => $e->getMessage(),
                    'timestamp' => now()->format('c')
                ]
            ], 500);
        }

  /**
 * Store a newly created info (POST /api/infos) WITH STOCK CHECK
 */
public function store(Request $request, StockService $stockService): JsonResponse
{
    try {
        Log::info('=== INFO STORE START ===');

        // Validation
        $validated = $request->validate([
            'customer_name' => 'required|string|max:255',
            'customer_email' => 'required|email',
            'customer_phone' => 'required|string',
            'customer_cpf' => 'required|string',
            'total' => 'required|numeric|min:0',
            'subtotal' => 'required|numeric|min:0',
            'shipping' => 'required|numeric|min:0',
            'discount' => 'sometimes|numeric|min:0',
            'payment_method' => 'required|string|in:pix,credit,debit,boleto',
            'payment_status' => 'required|string|in:pending,paid,failed,refunded',
            'status' => 'required|string|in:pending,processing,shipped,delivered,cancelled',
            'shipping_street' => 'required|string',
            'shipping_number' => 'required|string',
            'shipping_complement' => 'nullable|string',
            'shipping_neighborhood' => 'required|string',
            'shipping_city' => 'required|string',
            'shipping_state' => 'required|string|size:2',
            'shipping_zip_code' => 'required|string',
            'items' => 'required|array|min:1',
            'items.*.product_name' => 'required|string', // ✅ PRODUCT NAME IS REQUIRED
            'items.*.quantity' => 'required|integer|min:1',
            'items.*.price' => 'required|numeric|min:0',
            'items.*.total' => 'required|numeric|min:0',
            'items.*.image' => 'sometimes|string|nullable',
            'items.*.color' => 'sometimes|string|nullable',
            'items.*.size' => 'sometimes|string|nullable',
        ]);

        // ✅ ENSURE PRODUCT NAME IS PRESENT IN EACH ITEM
        $itemsWithProductName = [];
        foreach ($validated['items'] as $item) {
            if (empty($item['product_name'])) {
                return response()->json([
                    'success' => false,
                    'message' => 'Product name is required for all items'
                ], 400);
            }
            
            // ✅ Add product_name explicitly to each item
            $itemsWithProductName[] = [
                'product_name' => $item['product_name'], // ✅ ENSURED
                'quantity' => $item['quantity'],
                'price' => $item['price'],
                'total' => $item['total'],
                'image' => $item['image'] ?? null,
                'color' => $item['color'] ?? null,
                'size' => $item['size'] ?? null,
            ];
        }

        // ✅ OPTIONAL: Check stock before creating order
        if ($request->has('check_stock') && $request->check_stock == true) {
            $stockCheck = $stockService->checkStockAvailability($itemsWithProductName);
            $unavailableItems = array_filter($stockCheck, function($result) {
                return $result['available'] === false;
            });
            
            if (!empty($unavailableItems)) {
                return response()->json([
                    'success' => false,
                    'message' => 'Some items are out of stock',
                    'stock_check' => $stockCheck,
                    'unavailable_items' => $unavailableItems
                ], 400);
            }
        }

        // Generate order number
        $orderNumber = $request->order_number ?? 'ORD-' . date('Ymd') . '-' . strtoupper(substr(uniqid(), -6));

        // Prepare the data
        $infoData = [
            'order_number' => $orderNumber,
            'customer_name' => $validated['customer_name'],
            'customer_email' => $validated['customer_email'],
            'customer_phone' => $validated['customer_phone'],
            'customer_cpf' => $validated['customer_cpf'],
            'total' => $validated['total'],
            'subtotal' => $validated['subtotal'],
            'shipping' => $validated['shipping'],
            'discount' => $validated['discount'] ?? 0,
            'payment_method' => $validated['payment_method'],
            'payment_status' => $validated['payment_status'],
            'stock_updated' => false,
            'stock_updated_at' => null,
            'status' => $validated['status'],
            'shipping_street' => $validated['shipping_street'],
            'shipping_number' => $validated['shipping_number'],
            'shipping_complement' => $validated['shipping_complement'] ?? null,
            'shipping_neighborhood' => $validated['shipping_neighborhood'],
            'shipping_city' => $validated['shipping_city'],
            'shipping_state' => $validated['shipping_state'],
            'shipping_zip_code' => $validated['shipping_zip_code'],
            'items' => json_encode($itemsWithProductName), // ✅ NOW INCLUDES PRODUCT NAME
            'metadata' => $request->metadata ? json_encode($request->metadata) : null,
        ];

        Log::info('Creating order with items:', [
            'order_number' => $orderNumber,
            'items' => $itemsWithProductName
        ]);

        $info = Info::create($infoData);

        // ✅ If payment is already marked as paid, deduct stock immediately
        if ($validated['payment_status'] === 'paid') {
            $stockUpdated = $stockService->deductStockFromOrder($info);
            if (!$stockUpdated) {
                $info->delete();
                return response()->json([
                    'success' => false,
                    'message' => 'Failed to create order. Insufficient stock available.',
                    'order_number' => $orderNumber
                ], 400);
            }
        }

        return response()->json([
            'success' => true,
            'message' => 'Order created successfully',
            'data' => [
                'order_id' => $info->id,
                'order_number' => $info->order_number,
                'items' => $info->parsed_items, // ✅ WILL SHOW PRODUCT NAMES
                'stock_updated' => $info->stock_updated

    /**
     * ✅ Check stock availability before placing order
     * POST /api/infos/check-stock
     */
    public function checkStock(Request $request, StockService $stockService): JsonResponse
    {
        try {
            $validated = $request->validate([
                'items' => 'required|array|min:1',
                'items.*.product_name' => 'required|string',
                'items.*.quantity' => 'required|integer|min:1',
                'items.*.color' => 'sometimes|string|nullable',
                'items.*.size' => 'sometimes|string|nullable',
            ]);

            $stockCheck = $stockService->checkStockAvailability($validated['items']);
            
            $allAvailable = true;
            $messages = [];
            $unavailableItems = [];
            
            foreach ($stockCheck as $key => $result) {
                if (!$result['available']) {
                    $allAvailable = false;
                    $item = $validated['items'][$key];
                    $messages[] = "Estoque insuficiente para {$item['product_name']}: Disponível {$result['available_stock']}, Solicitado {$item['quantity']}";
                    $unavailableItems[] = [
                        'product_name' => $item['product_name'],
                        'requested_quantity' => $item['quantity'],
                        'available_stock' => $result['available_stock'],
                        'color' => $item['color'] ?? null,
                        'size' => $item['size'] ?? null
                    ];
                }
            }

            return response()->json([
                'success' => true,
                'all_available' => $allAvailable,
                'stock_check' => $stockCheck,
                'messages' => $messages,
                'unavailable_items' => $unavailableItems,
                'debug' => [
                    'items_count' => count($validated['items']),
                    'timestamp' => now()->format('c')
                ]
            ]);

        } catch (Exception $e) {
            Log::error('Check stock error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Failed to check stock',
                'error' => $e->getMessage()
            ], 500);
        }
    }

    ];

    /**
     * Get grouped stock by product
     */
    public static function getGroupedStock()
    {
        if (!Schema::hasTable('stock')) {
            return collect();
        }

        return self::orderBy('product_id')
            ->orderBy('color')
            ->orderBy('size')
            ->get()
            ->groupBy('product_id')
            ->map(function ($items, $productId) {
                $firstItem = $items->first();
                
                return [
                    'id' => $firstItem->id,
                    'product_id' => $productId,
                    'name' => $firstItem->name,
                    'category' => $firstItem->category,
                    'supplier' => $firstItem->supplier,
                    'cost' => (float) $firstItem->cost,
                    'profit' => (float) $firstItem->profit,
                    'sale_price' => (float) $firstItem->sale_price,
                    'total_value' => (float) $firstItem->total_value,
                    'last_updated' => $firstItem->last_updated,
                    'created_at' => $firstItem->created_at?->format('Y-m-d H:i:s'),
                    'updated_at' => $firstItem->updated_at?->format('Y-m-d H:i:s'),
                    'variants' => $items->map(function ($item) {
                        return [
                            'id' => $item->id,
                            'color' => $item->color,
                            'size' => $item->size,
                            'stock' => $item->stock, // ✅ AVAILABLE QUANTITY
                            'sku' => $item->sku,
                        ];
                    })->values()->toArray()
                ];
            })->values();
    }

    public static function getLowStock()
    {
        if (!Schema::hasTable('stock')) {
            return collect();
        }

        return self::where('stock', '>', 0)
            ->where('stock', '<=', 5)
            ->get();
    }

    public static function getOutOfStock()
    {
        if (!Schema::hasTable('stock')) {
            return collect();
        }

        return self::where('stock', 0)->get();
    }

    /**
     * Update specific field for a product (supplier, category, etc.)
     */
    public function updateProductField(Request $request, $productId): JsonResponse
    {
        try {
            $validator = Validator::make($request->all(), [
                'product_id' => 'sometimes|string',
                'name' => 'sometimes|string',
                'sku' => 'sometimes|string',
                'category' => 'sometimes|string',
                'supplier' => 'sometimes|string'
            ]);

            if ($validator->fails()) {
                return response()->json([
                    'success' => false,
                    'message' => 'Validation error',
                    'errors' => $validator->errors()
                ], 422);
            }

            // Check if any stock items exist with this product_id
            $existingItems = Stock::where('product_id', $productId)->get();
            
            if ($existingItems->isEmpty()) {
                return response()->json([
                    'success' => false,
                    'message' => 'Nenhum produto encontrado com esse product_id'
                ], 404);
            }

            $updateData = ['last_updated' => now()->format('Y-m-d')];

            // Only update fields that are provided
            if ($request->has('product_id')) {
                $updateData['product_id'] = $request->product_id;
            }
            if ($request->has('name')) {
                $updateData['name'] = $request->name;
            }
            if ($request->has('sku')) {
                $updateData['sku'] = $request->sku;
            }
            if ($request->has('category')) {
                $updateData['category'] = $request->category;
            }
            if ($request->has('supplier')) {
                $updateData['supplier'] = $request->supplier;
            }

            $updated = Stock::where('product_id', $productId)->update($updateData);

            return response()->json([
                'success' => true,
                'message' => 'Campo atualizado com sucesso em todas as variantes',
                'updated_count' => $updated,
                'updated_fields' => array_keys($updateData)
            ]);

        } catch (\Exception $e) {
            Log::error('Update product field error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Erro ao atualizar campo: ' . $e->getMessage()
            ], 500);
        }
    }

    public function updateProductPricing(Request $request, $productId): JsonResponse
    {
        try {
            $validator = Validator::make($request->all(), [
                'cost' => 'required|numeric|min:0',
                'profit' => 'required|numeric|min:0',
                'sale_price' => 'required|numeric|min:0',
                'total_value' => 'sometimes|numeric|min:0'
            ]);

            if ($validator->fails()) {
                return response()->json([
                    'success' => false,
                    'message' => 'Validation error',
                    'errors' => $validator->errors()
                ], 422);
            }

            $updated = Stock::where('product_id', $productId)
                ->update([
                    'cost' => $request->cost,
                    'profit' => $request->profit,
                    'sale_price' => $request->sale_price,
                    'total_value' => $request->total_value ?? DB::raw('stock * sale_price'),
                    'last_updated' => now()->format('Y-m-d')
                ]);

            if ($updated === 0) {
                return response()->json([
                    'success' => false,
                    'message' => 'Nenhum produto encontrado com esse product_id'
                ], 404);
            }

            return response()->json([
                'success' => true,
                'message' => 'Preços atualizados com sucesso em todas as variantes',
                'updated_count' => $updated
            ]);

        } catch (\Exception $e) {
            Log::error('Update pricing error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Erro ao atualizar preços: ' . $e->getMessage()
            ], 500);
        }
    }

    public function updateStock(Request $request, $id): JsonResponse
    {
        try {
            $validator = Validator::make($request->all(), [
                'stock' => 'required|integer|min:0'
            ]);

            if ($validator->fails()) {
                return response()->json([
                    'success' => false,
                    'message' => 'Validation error',
                    'errors' => $validator->errors()
                ], 422);
            }

            $stock = Stock::find($id);
            
            if (!$stock) {
                return response()->json([
                    'success' => false,
                    'message' => 'Stock item not found'
                ], 404);
            }

            $stock->update([
                'stock' => $request->stock,
                'last_updated' => now()->format('Y-m-d')
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Stock updated successfully',
                'data' => $stock
            ]);

        } catch (\Exception $e) {
            Log::error('Update stock error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Failed to update stock'
            ], 500);
        }
    }

    public function store(Request $request): JsonResponse
    {
        try {
            $validator = Validator::make($request->all(), [
                'product_id' => 'required|string',
                'name' => 'required|string',
                'sku' => 'required|string|unique:stock,sku',
                'category' => 'required|string',
                'supplier' => 'required|string',
                'cost' => 'required|numeric|min:0',
                'profit' => 'required|numeric|min:0',
                'sale_price' => 'required|numeric|min:0',
                'total_value' => 'required|numeric|min:0',
                'color' => 'required|string',
                'size' => 'required|string',
                'stock' => 'required|integer|min:0'
            ]);

            if ($validator->fails()) {
                return response()->json([
                    'success' => false,
                    'message' => 'Validation error',
                    'errors' => $validator->errors()
                ], 422);
            }

            $stock = Stock::create([
                'product_id' => $request->product_id,
                'name' => $request->name,
                'sku' => $request->sku,
                'category' => $request->category,
                'supplier' => $request->supplier,
                'cost' => $request->cost,
                'profit' => $request->profit,
                'sale_price' => $request->sale_price,
                'total_value' => $request->total_value,
                'color' => $request->color,
                'size' => $request->size,
                'stock' => $request->stock,
                'last_updated' => now()->format('Y-m-d')
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Stock item created successfully',
                'data' => $stock
            ], 201);

        } catch (\Exception $e) {
            Log::error('Create stock item error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Failed to create stock item',
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Delete a specific stock item (variant)
     */
    public function destroy($id): JsonResponse
    {
        try {
            Log::info("Attempting to delete stock item with ID: {$id}");

            $stock = Stock::find($id);
            
            if (!$stock) {
                Log::warning("Stock item not found with ID: {$id}");
                return response()->json([
                    'success' => false,
                    'message' => 'Item de estoque não encontrado'
                ], 404);
            }

            // Store info for logging before deletion
            $productInfo = [
                'id' => $stock->id,
                'product_id' => $stock->product_id,
                'name' => $stock->name,
                'color' => $stock->color,
                'size' => $stock->size
            ];

            $stock->delete();

            Log::info("Stock item deleted successfully", $productInfo);

            return response()->json([
                'success' => true,
                'message' => 'Variante excluída com sucesso',
                'deleted_item' => $productInfo
            ]);

        } catch (\Exception $e) {
            Log::error('Delete stock item error: ' . $e->getMessage(), [
                'stock_id' => $id,
                'exception' => $e->getTraceAsString()
            ]);
            
            return response()->json([
                'success' => false,
                'message' => 'Erro ao excluir variante: ' . $e->getMessage()
            ], 500);
        }
    }

    /**
     * Delete all variants of a product by product_id
     */
    public function destroyByProductId($productId): JsonResponse
    {
        try {
            Log::info("Attempting to delete all stock items for product_id: {$productId}");

            // First, check if any items exist with this product_id
            $existingItems = Stock::where('product_id', $productId)->get();
            
            if ($existingItems->isEmpty()) {
                Log::warning("No stock items found for product_id: {$productId}");
                return response()->json([
                    'success' => false,
                    'message' => 'Nenhum produto encontrado com este ID'
                ], 404);
            }

            // Store info for logging before deletion
            $itemsInfo = $existingItems->map(function ($item) {
                return [
                    'id' => $item->id,
                    'product_id' => $item->product_id,
                    'name' => $item->name,
                    'color' => $item->color,
                    'size' => $item->size
                ];
            })->toArray();

            $deletedCount = Stock::where('product_id', $productId)->delete();

            Log::info("Product and all variants deleted successfully", [
                'product_id' => $productId,
                'deleted_count' => $deletedCount,
                'deleted_items' => $itemsInfo
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Produto e todas as variantes excluídos com sucesso',
                'deleted_count' => $deletedCount,
                'deleted_items' => $itemsInfo
            ]);

        } catch (\Exception $e) {
            Log::error('Delete product error: ' . $e->getMessage(), [
                'product_id' => $productId,
                'exception' => $e->getTraceAsString()
            ]);
            
            return response()->json([
                'success' => false,
                'message' => 'Erro ao excluir produto: ' . $e->getMessage()
            ], 500);
        }
    }
