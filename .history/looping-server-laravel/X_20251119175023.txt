Make it appear color and size on Order tab of my admin panel 

OrdersManagement.jsx 

import React, { useState, useEffect } from "react";
import {
  FiSearch,
  FiEye,
  FiRefreshCw,
  FiPackage,
  FiXCircle,
  FiTrash2,
  FiServer,
  FiChevronUp,
  FiChevronDown,
} from "react-icons/fi";

const OrdersManagement = () => {
  const [orderSearch, setOrderSearch] = useState("");
  const [orders, setOrders] = useState([]);
  const [loading, setLoading] = useState(true);
  const [selectedOrder, setSelectedOrder] = useState(null);
  const [showOrderModal, setShowOrderModal] = useState(false);
  const [deleteConfirm, setDeleteConfirm] = useState(null);
  const [error, setError] = useState(null);
  const [apiStatus, setApiStatus] = useState(null);
  
  const [sortConfig, setSortConfig] = useState({
    key: 'created_at',
    direction: 'desc'
  });

  const API_BASE_URL = 'http://localhost:8000/api';

  const sortableFields = {
    order_number: { label: 'Número do Pedido', type: 'string' },
    customer_name: { label: 'Nome do Cliente', type: 'string' },
    customer_email: { label: 'Email', type: 'string' },
    created_at: { label: 'Data do Pedido', type: 'date' },
    total: { label: 'Valor Total', type: 'number' },
    payment_method: { label: 'Método de Pagamento', type: 'string' },
    status: { label: 'Status', type: 'string' }
  };

  const apiService = {
    async getOrders() {
      try {
        console.log('Buscando pedidos em:', `${API_BASE_URL}/infos`);
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 10000);

        const response = await fetch(`${API_BASE_URL}/infos`, {
          method: 'GET',
          headers: {
            'Accept': 'application/json',
            'Content-Type': 'application/json',
          },
          signal: controller.signal,
        });

        clearTimeout(timeoutId);

        const contentType = response.headers.get('content-type');
        if (contentType && contentType.includes('text/html')) {
          const text = await response.text();
          throw new Error('Erro no servidor. Verifique os logs do Laravel.');
        }

        if (!response.ok) {
          let errorMessage = `HTTP ${response.status}: ${response.statusText}`;
          try {
            const errorData = await response.json();
            errorMessage = errorData.message || errorData.error || errorMessage;
          } catch {}
          throw new Error(errorMessage);
        }

        const data = await response.json();
        if (!data.success) throw new Error(data.message || 'Erro na API');
        
        return data.data || [];
      } catch (error) {
        if (error.name === 'AbortError') throw new Error('Timeout: servidor demorou');
        if (error.message.includes('Failed to fetch')) throw new Error('Servidor offline ou CORS bloqueado');
        throw error;
      }
    },

    async generateSampleOrders() {
      const response = await fetch(`${API_BASE_URL}/infos/generate-samples`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
        body: JSON.stringify({}),
      });
      if (!response.ok) throw new Error(`Erro ${response.status}`);
      const data = await response.json();
      return data.data;
    },
  };

  const sortOrders = (orders, config) => {
    if (!config.key) return orders;

    return [...orders].sort((a, b) => {
      let aValue = a[config.key];
      let bValue = b[config.key];

      if (aValue == null) aValue = '';
      if (bValue == null) bValue = '';

      const fieldConfig = sortableFields[config.key];
      
      let comparison = 0;
      
      switch (fieldConfig?.type) {
        case 'number':
          aValue = parseFloat(aValue) || 0;
          bValue = parseFloat(bValue) || 0;
          comparison = aValue - bValue;
          break;
        
        case 'date':
          aValue = new Date(aValue || 0);
          bValue = new Date(bValue || 0);
          comparison = aValue - bValue;
          break;
        
        case 'string':
        default:
          aValue = String(aValue).toLowerCase();
          bValue = String(bValue).toLowerCase();
          comparison = aValue.localeCompare(bValue, 'pt-BR');
          break;
      }

      return config.direction === 'desc' ? -comparison : comparison;
    });
  };

  const handleSort = (key) => {
    setSortConfig(currentConfig => ({
      key,
      direction: 
        currentConfig.key === key && currentConfig.direction === 'asc' 
          ? 'desc' 
          : 'asc'
    }));
  };

  const renderSortIcon = (key) => {
    if (sortConfig.key !== key) {
      return <span className="text-gray-400">↕</span>;
    }
    
    return sortConfig.direction === 'asc' 
      ? <FiChevronUp className="text-black" size={16} />
      : <FiChevronDown className="text-black" size={16} />;
  };

  const getItemAttributes = (item) => {
    console.log('🔍 Estrutura completa do item:', JSON.stringify(item, null, 2));
    
    const size = 
      item.size ||
      item.selectedSize || 
      item.tamanho || 
      item.variant_size ||
      item.product_variant?.size ||
      item.variant?.size ||
      item.product?.size ||
      (item.options && (item.options.size || item.options.tamanho)) ||
      item.attributes?.size ||
      item.metadata?.size ||
      'Não especificado';
    
    const color = 
      item.color ||
      item.selectedColor || 
      item.cor || 
      item.variant_color ||
      item.product_variant?.color ||
      item.variant?.color ||
      item.product?.color ||
      (item.options && (item.options.color || item.options.cor)) ||
      item.attributes?.color ||
      item.metadata?.color ||
      'Não especificada';
    
    console.log(`📦 Item: ${item.product_name} | Tamanho: ${size} | Cor: ${color}`);
    return { size, color };
  };

  const getColorName = (color) => {
    if (!color || color === 'Não especificada' || color === 'undefined') {
      return 'Não especificada';
    }
    
    const colorMap = {
      "#000000": "Preto",
      "#000": "Preto",
      "#FFFFFF": "Branco", 
      "#fff": "Branco",
      "#1E90FF": "Azul",
      "#FF69B4": "Rosa",
      "#FF0000": "Vermelho",
      "#00FF00": "Verde",
      "#0000FF": "Azul Escuro",
      "#FFFF00": "Amarelo",
      "#FFA500": "Laranja",
      "#800080": "Roxo",
      "#808080": "Cinza",
      "#964B00": "Marrom",
      
      "preto": "Preto",
      "branco": "Branco", 
      "azul": "Azul",
      "rosa": "Rosa",
      "vermelho": "Vermelho",
      "verde": "Verde",
      "amarelo": "Amarelo",
      "laranja": "Laranja",
      "roxo": "Roxo",
      "cinza": "Cinza",
      "marrom": "Marrom",
      
      "black": "Preto",
      "white": "Branco",
      "blue": "Azul",
      "pink": "Rosa", 
      "red": "Vermelho",
      "green": "Verde",
      "yellow": "Amarelo",
      "orange": "Laranja",
      "purple": "Roxo",
      "gray": "Cinza",
      "grey": "Cinza",
      "brown": "Marrom"
    };
    
    const normalizedColor = color.toString().toLowerCase().trim().replace(/\s+/g, '');
    
    if (colorMap[normalizedColor]) {
      return colorMap[normalizedColor];
    }
    
    for (const [key, value] of Object.entries(colorMap)) {
      if (normalizedColor.includes(key.toLowerCase()) || key.toLowerCase().includes(normalizedColor)) {
        return value;
      }
    }
    
    if (/^#([A-Fa-f0-9]{6}|[A-Fa-f0-9]{3})$/.test(color)) {
      return color.toUpperCase();
    }
    
    return color.charAt(0).toUpperCase() + color.slice(1);
  };

  const testApiConnection = async () => {
    setLoading(true);
    try {
      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), 5000);
      const response = await fetch(`${API_BASE_URL}/test`, {
        signal: controller.signal,
      });
      clearTimeout(timeoutId);

      const result = response.ok
        ? { connected: true, data: await response.json() }
        : { connected: false, error: `HTTP ${response.status}` };

      setApiStatus(result);
      alert(result.connected ? 'API conectada!' : `Falha: ${result.error}`);
    } catch (err) {
      const msg = err.name === 'AbortError' ? 'Timeout no teste' : err.message;
      setApiStatus({ connected: false, error: msg });
      alert('Erro ao testar conexão');
    } finally {
      setLoading(false);
    }
  };

  const fetchOrders = async () => {
    setLoading(true);
    setError(null);
    try {
      const ordersData = await apiService.getOrders();
      setOrders(ordersData);
    } catch (err) {
      setError("Erro ao carregar pedidos: " + err.message);
      setOrders([]);
    } finally {
      setLoading(false);
    }
  };

  const generateSampleOrders = async () => {
    if (!window.confirm("Deseja gerar pedidos de exemplo para teste?")) return;
    try {
      await apiService.generateSampleOrders();
      fetchOrders();
    } catch (err) {
      setError("Erro ao gerar exemplo: " + err.message);
    }
  };

  const updateOrderStatus = (orderId, newStatus) => {
    setOrders(prev => prev.map(o => o.id === orderId ? { ...o, status: newStatus } : o));
  };

  const deleteOrder = (orderId) => {
    setOrders(prev => prev.filter(o => o.id !== orderId));
    setDeleteConfirm(null);
  };

  const viewOrderDetails = (order) => {
    console.log('📋 Pedido selecionado para detalhes:', order);
    setSelectedOrder(order);
    setShowOrderModal(true);
  };

  const cancelOrder = (orderId) => {
    if (window.confirm("Tem certeza que deseja cancelar este pedido?")) {
      updateOrderStatus(orderId, "cancelled");
    }
  };

  const confirmDelete = (order) => setDeleteConfirm(order);
  const cancelDelete = () => setDeleteConfirm(null);

  const getStatusText = (status) => {
    const map = {
      pending: "Pendente",
      processing: "Em Preparação",
      shipped: "Enviado",
      delivered: "Entregue",
      cancelled: "Cancelado",
    };
    return map[status] || status;
  };

  const getPaymentText = (payment) => {
    const map = { 
      pix: "PIX", 
      credit_card: "Cartão de Crédito", 
      boleto: "Boleto",
      debit_card: "Cartão de Débito",
      cash: "Dinheiro"
    };
    return map[payment] || payment;
  };

  const getStatusColor = (status) => {
    const colors = {
      shipped: "bg-green-100 text-green-800 border border-green-200",
      processing: "bg-yellow-100 text-yellow-800 border border-yellow-200",
      pending: "bg-blue-100 text-blue-800 border border-blue-200",
      delivered: "bg-purple-100 text-purple-800 border border-purple-200",
      cancelled: "bg-red-100 text-red-800 border border-red-200",
    };
    return colors[status] || "bg-gray-100 text-gray-800 border border-gray-200";
  };

  const getPaymentColor = (payment) => {
    const colors = {
      credit_card: "bg-green-100 text-green-800",
      debit_card: "bg-blue-100 text-blue-800",
      pix: "bg-blue-100 text-blue-800",
      boleto: "bg-purple-100 text-purple-800",
      cash: "bg-gray-100 text-gray-800",
    };
    return colors[payment] || "bg-gray-100 text-gray-800";
  };

  const formatCurrency = (value) => {
    if (value == null || isNaN(value)) return 'R$ 0,00';
    return `R$ ${parseFloat(value).toFixed(2)}`;
  };

  const filteredAndSortedOrders = sortOrders(
    orders.filter(order =>
      order && (
        (order.order_number?.toLowerCase().includes(orderSearch.toLowerCase())) ||
        (order.customer_name?.toLowerCase().includes(orderSearch.toLowerCase())) ||
        (order.customer_email?.toLowerCase().includes(orderSearch.toLowerCase())) ||
        getStatusText(order.status).toLowerCase().includes(orderSearch.toLowerCase()) ||
        getPaymentText(order.payment_method).toLowerCase().includes(orderSearch.toLowerCase()) ||
        formatCurrency(order.total).toLowerCase().includes(orderSearch.toLowerCase())
      )
    ),
    sortConfig
  );

  const getDisplayValue = (order, key) => {
    const value = order[key];
    
    switch (key) {
      case 'total':
        return formatCurrency(value);
      case 'created_at':
        return order.created_at ? new Date(order.created_at).toLocaleDateString('pt-BR') : 'N/A';
      case 'status':
        return getStatusText(value);
      case 'payment_method':
        return getPaymentText(value);
      default:
        return value || 'N/A';
    }
  };

  useEffect(() => {
    fetchOrders();
  }, []);

  return (
    <div className="p-6 space-y-6" style={{ fontFamily: 'Calibri, sans-serif' }}>
      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
        <div>
          <h2 className="text-xl font-semibold">Gerenciamento de Pedidos</h2>
          <p className="text-gray-600 text-sm mt-1">
            {filteredAndSortedOrders.length} {filteredAndSortedOrders.length === 1 ? 'pedido' : 'pedidos'} encontrados
            {filteredAndSortedOrders.length > 0 && (
              <span className="text-blue-600 ml-2">
                • Ordenado por: {sortableFields[sortConfig.key]?.label} ({sortConfig.direction === 'asc' ? 'crescente' : 'decrescente'})
              </span>
            )}
          </p>
        </div>

        <div className="flex items-center gap-3 flex-wrap">
          <button onClick={fetchOrders} className="flex items-center gap-2 bg-blue-600 hover:bg-blue-700 text-white px-4 py-2 rounded-lg text-sm font-medium transition-colors">
            <FiRefreshCw size={16} /> Atualizar
          </button>
          <button onClick={testApiConnection} className="flex items-center gap-2 bg-green-600 hover:bg-green-700 text-white px-4 py-2 rounded-lg text-sm font-medium transition-colors">
            <FiServer size={16} /> Testar API
          </button>
          <button onClick={generateSampleOrders} className="flex items-center gap-2 bg-green-600 hover:bg-green-700 text-white px-4 py-2 rounded-lg text-sm font-medium transition-colors">
            <FiPackage size={16} /> Gerar Exemplo
          </button>
          <div className="relative w-full sm:w-64">
            <input
              type="text"
              placeholder="Buscar pedidos..."
              value={orderSearch}
              onChange={(e) => setOrderSearch(e.target.value)}
              className="w-full p-2 pl-10 border border-gray-300 rounded text-black focus:outline-none focus:ring-2 focus:ring-blue-500"
            />
            <FiSearch className="absolute left-3 top-2.5 text-gray-400" size={16} />
          </div>
        </div>
      </div>

      <div className="bg-gray-50 p-4 rounded-lg border">
        <div className="flex flex-wrap items-center gap-2">
          <span className="text-sm font-medium text-gray-700">Ordenar por:</span>
          {Object.entries(sortableFields).map(([key, config]) => (
            <button
              key={key}
              onClick={() => handleSort(key)}
              className={`
                box-border inline-flex items-center gap-1
                px-3 py-1 rounded text-xs font-medium transition-colors border
                focus:outline-none focus:ring-0
                ${
                  sortConfig.key === key
                    ? "bg-blue-600 text-white border-blue-600"
                    : "bg-white text-gray-700 border-gray-300 hover:bg-gray-50"
                }
              `}
            >
              <span className="leading-none">{config.label}</span>
              {renderSortIcon(key)}
            </button>
          ))}

          <button
            onClick={() => setSortConfig({ key: 'created_at', direction: 'desc' })}
            className="box-border inline-flex items-center gap-1 px-3 py-1 rounded text-xs font-medium border bg-gray-200 text-gray-700 border-gray-300 hover:bg-gray-300 focus:outline-none focus:ring-0"
          >
            Resetar
          </button>
        </div>
      </div>

      <div className="bg-white border border-gray-200 rounded-lg overflow-hidden">
        <div className="overflow-x-auto">
          <table className="min-w-full divide-y divide-gray-200">
            <thead className="bg-gray-50">
              <tr>
                <th 
                  className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider cursor-pointer hover:bg-gray-100 transition-colors"
                  onClick={() => handleSort('order_number')}
                >
                  <div className="flex items-center gap-1">
                    Pedido
                    {renderSortIcon('order_number')}
                  </div>
                </th>
                <th 
                  className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider cursor-pointer hover:bg-gray-100 transition-colors"
                  onClick={() => handleSort('customer_name')}
                >
                  <div className="flex items-center gap-1">
                    Cliente
                    {renderSortIcon('customer_name')}
                  </div>
                </th>
                <th 
                  className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider cursor-pointer hover:bg-gray-100 transition-colors"
                  onClick={() => handleSort('created_at')}
                >
                  <div className="flex items-center gap-1">
                    Data
                    {renderSortIcon('created_at')}
                  </div>
                </th>
                <th 
                  className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider cursor-pointer hover:bg-gray-100 transition-colors"
                  onClick={() => handleSort('total')}
                >
                  <div className="flex items-center gap-1">
                    Total
                    {renderSortIcon('total')}
                  </div>
                </th>
                <th 
                  className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider cursor-pointer hover:bg-gray-100 transition-colors"
                  onClick={() => handleSort('payment_method')}
                >
                  <div className="flex items-center gap-1">
                    Pagamento
                    {renderSortIcon('payment_method')}
                  </div>
                </th>
                <th 
                  className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider cursor-pointer hover:bg-gray-100 transition-colors"
                  onClick={() => handleSort('status')}
                >
                  <div className="flex items-center gap-1">
                    Status
                    {renderSortIcon('status')}
                  </div>
                </th>
                <th className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">
                  Ações
                </th>
              </tr>
            </thead>
            <tbody className="bg-white divide-y divide-gray-200">
              {filteredAndSortedOrders.map((order) => (
                <tr key={order.id} className="hover:bg-gray-50">
                  <td className="px-4 sm:px-6 py-4 whitespace-nowrap text-sm font-medium text-gray-900">
                    {getDisplayValue(order, 'order_number')}
                  </td>
                  <td className="px-4 sm:px-6 py-4 whitespace-nowrap">
                    <div className="text-sm text-gray-900">{getDisplayValue(order, 'customer_name')}</div>
                    <div className="text-sm text-gray-500">{order.customer_email || 'N/A'}</div>
                  </td>
                  <td className="px-4 sm:px-6 py-4 whitespace-nowrap text-sm text-gray-900">
                    {getDisplayValue(order, 'created_at')}
                    {order.created_at && (
                      <div className="text-xs text-gray-500">
                        {new Date(order.created_at).toLocaleTimeString('pt-BR')}
                      </div>
                    )}
                  </td>
                  <td className="px-4 sm:px-6 py-4 whitespace-nowrap text-sm font-semibold text-gray-900">
                    {getDisplayValue(order, 'total')}
                  </td>
                  <td className="px-4 sm:px-6 py-4 whitespace-nowrap">
                    {/* ✅ PAGAMENTO BUTTON - SQUARED, NO ROUNDNESS, VERTICALLY ALIGNED */}
                    <span className={`inline-flex items-center px-2 py-1 rounded text-xs font-medium ${getPaymentColor(order.payment_method)}`}>
                      {getDisplayValue(order, 'payment_method')}
                    </span>
                  </td>
                  <td className="px-4 sm:px-6 py-4 whitespace-nowrap">
                    {/* ✅ STATUS BUTTON - SQUARED, NO ROUNDNESS, VERTICALLY ALIGNED */}
                    <select
                      value={order.status || 'pending'}
                      onChange={(e) => updateOrderStatus(order.id, e.target.value)}
                      className={`text-xs font-medium px-3 py-1 border-0 focus:ring-2 rounded focus:ring-blue-500 ${getStatusColor(order.status)}`}
                    >
                      <option value="pending">Pendente</option>
                      <option value="processing">Em Preparação</option>
                      <option value="shipped">Enviado</option>
                      <option value="delivered">Entregue</option>
                      <option value="cancelled">Cancelado</option>
                    </select>
                  </td>
                  <td className="px-4 sm:px-6 py-4 whitespace-nowrap text-sm font-medium space-x-2">
                    <button onClick={() => viewOrderDetails(order)} className="text-indigo-600 hover:text-indigo-900 p-1 rounded transition-colors" title="Ver Detalhes">
                      <FiEye size={16} />
                    </button>
                    <button onClick={() => cancelOrder(order.id)} className="text-yellow-600 hover:text-yellow-900 p-1 rounded transition-colors" title="Cancelar Pedido">
                      <FiXCircle size={16} />
                    </button>
                    <button onClick={() => confirmDelete(order)} className="text-red-600 hover:text-red-900 p-1 rounded transition-colors" title="Excluir Pedido">
                      <FiTrash2 size={16} />
                    </button>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        </div>
      </div>

      {showOrderModal && selectedOrder && (
        <div className="fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center p-4 z-50">
          <div className="bg-white rounded-lg max-w-4xl w-full max-h-[90vh] overflow-y-auto">
            <div className="p-6 border-b border-gray-200">
              <div className="flex justify-between items-center">
                <h3 className="text-lg font-semibold">Detalhes do Pedido - {selectedOrder.order_number}</h3>
                <button onClick={() => setShowOrderModal(false)} className="text-gray-400 hover:text-gray-600">
                  <FiXCircle size={20} />
                </button>
              </div>
            </div>

            <div className="p-6 space-y-6">
              <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
                <div>
                  <h4 className="font-semibold text-gray-900 mb-3">Informações do Pedido</h4>
                  <div className="space-y-2">
                    <p><strong>Número:</strong> {selectedOrder.order_number || 'N/A'}</p>
                    <p><strong>Data:</strong> {selectedOrder.created_at ? new Date(selectedOrder.created_at).toLocaleString('pt-BR') : 'N/A'}</p>
                    <p><strong>Status:</strong> <span className={`px-2 py-1 text-xs ${getStatusColor(selectedOrder.status)}`}>{getStatusText(selectedOrder.status)}</span></p>
                    <p><strong>Total:</strong> {formatCurrency(selectedOrder.total)}</p>
                    <p><strong>Subtotal:</strong> {formatCurrency(selectedOrder.subtotal)}</p>
                    <p><strong>Frete:</strong> {formatCurrency(selectedOrder.shipping)}</p>
                    <p><strong>Desconto:</strong> {formatCurrency(selectedOrder.discount)}</p>
                  </div>
                </div>

                <div>
                  <h4 className="font-semibold text-gray-900 mb-3">Informações do Cliente</h4>
                  <div className="space-y-2">
                    <p><strong>Nome:</strong> {selectedOrder.customer_name || 'N/A'}</p>
                    <p><strong>Email:</strong> {selectedOrder.customer_email || 'N/A'}</p>
                    <p><strong>Telefone:</strong> {selectedOrder.customer_phone || 'Não informado'}</p>
                    <p><strong>CPF:</strong> {selectedOrder.customer_cpf || 'Não informado'}</p>
                    <p><strong>Pagamento:</strong> <span className={`px-2 py-1 text-xs ${getPaymentColor(selectedOrder.payment_method)}`}>{getPaymentText(selectedOrder.payment_method)}</span></p>
                    <p><strong>Status Pagamento:</strong> {selectedOrder.payment_status || 'N/A'}</p>
                  </div>
                </div>
              </div>

              <div>
                <h4 className="font-semibold text-gray-900 mb-3">Itens do Pedido ({selectedOrder.items?.length || 0})</h4>
                {selectedOrder.items?.length > 0 ? (

// ✅ SIMPLIFIED CSRF Token Management
const CsrfTokenManager = {
  getCsrfToken: async () => {
    try {
      console.log('🔄 Fetching CSRF token...');
      
      const response = await fetch('http://localhost:8000/sanctum/csrf-cookie', {
        method: 'GET',
        credentials: 'include',
      });

      if (response.ok) {
        const cookies = document.cookie;
        const tokenMatch = cookies.match(/XSRF-TOKEN=([^;]+)/);
        
        if (tokenMatch && tokenMatch[1]) {
          const token = decodeURIComponent(tokenMatch[1]);
          console.log('✅ CSRF token obtained');
          return token;
        }
      }
      
      const metaToken = document.querySelector('meta[name="csrf-token"]')?.getAttribute('content');
      return metaToken || 'fallback-csrf-token';
      
    } catch (error) {
      console.error('❌ Error getting CSRF token:', error);
      return 'fallback-csrf-token';
    }
  }
};

// ✅ SIMPLIFIED Backend API Service
const OrderAPI = {
  createOrder: async (orderData) => {
    try {
      const csrfToken = await CsrfTokenManager.getCsrfToken();
      
      const response = await fetch('http://localhost:8000/api/infos', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json',
          'X-XSRF-TOKEN': csrfToken,
        },
        credentials: 'include',
        body: JSON.stringify(orderData)
      });

      if (!response.ok) {
        throw new Error(`HTTP ${response.status}`);
      }

      const result = await response.json();
      
      if (!result.success) {
        throw new Error(result.message || 'Order creation failed');
      }
      
      return result;
    } catch (error) {
      console.error('❌ Order API Error:', error);
      throw error;
    }
  },

  updateOrderStatus: async (orderId, status, paymentStatus) => {
    try {
      const csrfToken = await CsrfTokenManager.getCsrfToken();
      
      const response = await fetch(`http://localhost:8000/api/infos/${orderId}/status`, {
        method: 'PUT',
        headers: {
          'Content-Type': 'application/json',
          'Accept': 'application/json',
          'X-XSRF-TOKEN': csrfToken,
        },
        credentials: 'include',
        body: JSON.stringify({
          status: status,
          payment_status: paymentStatus
        })
      });

      if (!response.ok) {
        throw new Error(`HTTP error! status: ${response.status}`);
      }

      const result = await response.json();
      
      if (!result.success) {
        throw new Error(result.message || 'Status update failed');
      }
      
      return result;
    } catch (error) {
      console.error('❌ Order Status Update Error:', error);
      throw error;
    }
  }
};

// ✅ CEP Input Component
const CepInput = ({ 
  cep, 
  onCepChange, 
  loading, 
  shippingData,
  onAutoFillAddress 
}) => {
  const [isValidCep, setIsValidCep] = useState(false);
  const [showError, setShowError] = useState(false);

  const handleCepChange = (e) => {
    const value = e.target.value.replace(/\D/g, '');
    let formattedCep = value;
    
    if (value.length > 5) {
      formattedCep = value.substring(0, 5) + '-' + value.substring(5, 8);
    }
    
    onCepChange(formattedCep);
    
    const hasValidLength = value.length === 8;
    setIsValidCep(hasValidLength);
    
    if (value.length > 0 && !hasValidLength && value.length < 8) {
      setShowError(false);
    } else {
      setShowError(value.length > 0 && !hasValidLength);
    }
  };

  const handleCepBlur = () => {
    const cleanCep = cep.replace(/\D/g, '');
    const hasValidLength = cleanCep.length === 8;
    
    setShowError(cleanCep.length > 0 && !hasValidLength);
    
    if (hasValidLength) {
      onAutoFillAddress(cep);
    }
  };

  const getMissingDigits = () => {
    const cleanCep = cep.replace(/\D/g, '');
    return 8 - cleanCep.length;
  };

  return (
    <div className="bg-white p-4 rounded-lg border border-gray-200 mb-6">
      <div className="flex items-center gap-2 mb-3">
        <FiMapPin className="text-blue-600" size={20} />
        <h3 className="text-lg font-semibold text-gray-800">Calcular Frete</h3>
      </div>
      
      <div className="flex gap-3">
        <div className="flex-1">
          <label className="block text-sm font-medium text-gray-700 mb-2">
            Digite seu CEP
          </label>
          <div className="relative">
            <input
              type="text"
              value={cep}
              onChange={handleCepChange}
              onBlur={handleCepBlur}
              className="w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent text-black pr-24"
              placeholder="00000-000"
              maxLength={9}
            />
            <div className="absolute right-2 top-1/2 transform -translate-y-1/2">
              {loading && (
                <div className="animate-spin rounded-full h-5 w-5 border-b-2 border-blue-600"></div>
              )}
            </div>
          </div>
          
          {showError && (
            <p className="text-red-500 text-xs mt-1">CEP deve ter 8 dígitos</p>
          )}
          
          {cep && cep.replace(/\D/g, '').length > 0 && cep.replace(/\D/g, '').length < 8 && !showError && (
            <p className="text-gray-500 text-xs mt-1">
              Digite mais {getMissingDigits()} dígito{getMissingDigits() !== 1 ? 's' : ''}
            </p>
          )}
        </div>
      </div>

      {shippingData.cep && isValidCep && (
        <div className="mt-3 p-3 bg-green-50 border border-green-200 rounded-lg">
          <div className="flex items-center gap-2">
            <FiCheck className="text-green-600" size={16} />
            <span className="text-green-800 text-sm font-medium">
              CEP {shippingData.cep} válido
            </span>
          </div>
          {shippingData.city && shippingData.state && (
            <p className="text-green-700 text-sm mt-1">
              {shippingData.city} - {shippingData.state}
            </p>
          )}
        </div>
      )}
    </div>
  );
};

          clearInterval(timer);

      <div className="flex items-center justify-between mb-4">
        <h2 className="text-xl font-semibold text-black">Resumo do Pedido</h2>
        <div className="flex items-center gap-2">
          {isUsingSelectedItems && (
            <span className="text-xs bg-green-100 text-green-800 px-2 py-1 rounded">
              {cartItems.length} {cartItems.length === 1 ? 'selecionado' : 'selecionados'}
            </span>
          )}
          <button 
            onClick={() => navigate('/cart')}
            className="text-xs bg-blue-100 text-blue-600 px-2 py-1 rounded hover:bg-blue-200 transition-colors"
            title="Alterar itens selecionados"
          >
            Alterar Itens
          </button>
        </div>
      </div>
      
      {isUsingSelectedItems && (
        <div className="mb-4 p-3 bg-green-50 border border-green-200 rounded-lg">
          <div className="flex items-center gap-2">
            <FiCheck className="text-green-600" size={16} />
            <span className="text-green-800 text-sm font-medium">
              {cartItems.length} {cartItems.length === 1 ? 'item selecionado' : 'itens selecionados'} do carrinho
            </span>
          </div>
        </div>
      )}
      
      <div className="space-y-3 mb-4 max-h-80 overflow-y-auto">
        {cartItems.map((item, index) => {
          const product = getProductDetails(item);
          const itemTotal = product.price * product.quantity;

          return (
            <div key={item.id || index} className="flex items-center space-x-3 pb-3 border-b border-gray-100 last:border-b-0">
              <div className="w-16 h-16 bg-gray-200 rounded-lg flex-shrink-0 flex items-center justify-center overflow-hidden">
                {product.image ? (
                  <img 
                    src={product.image} 
                    alt={product.name}
                    className="w-full h-full object-cover"
                  />
                ) : (
                  <span className="text-gray-500 text-xs">Imagem</span>
                )}
              </div>
              <div className="flex-1 min-w-0">
                <p className="font-medium text-gray-900 text-sm leading-tight">{product.name}</p>
                <p className="text-sm text-gray-600">Qtd: {product.quantity}</p>
                <p className="text-sm font-semibold text-gray-900">
                  R$ {product.price.toFixed(2)} cada
                </p>
              </div>
              <p className="font-semibold text-gray-900 text-sm">
                R$ {itemTotal.toFixed(2)}
              </p>
            </div>
          );
        })}
      </div>

      <div className="border-t pt-4 mb-4">
        <div className="flex justify-between items-center mb-2">
          <span className="text-sm text-gray-600">Itens no carrinho:</span>
          <span className="font-semibold text-black">{cartItems.length}</span>
        </div>
      </div>

      <div className="border-t pt-4 mb-4">
        <div className="flex space-x-2 mb-4">
          <input
            type="text"
            value={couponCode}
            onChange={(e) => setCouponCode(e.target.value)}
            placeholder="Cupom de desconto"
            className="flex-1 p-2 border border-gray-300 rounded-lg text-sm text-black"
          />
          <button
            onClick={applyCoupon}
            className="bg-gray-200 hover:bg-gray-300 text-gray-700 px-4 rounded-lg text-sm font-medium"
          >
            Aplicar
          </button>
        </div>
      </div>

      <div className="space-y-2 text-sm">
        <div className="flex justify-between text-black">
          <span>Subtotal</span>
          <span>R$ {calculateSubtotal().toFixed(2)}</span>
        </div>
        
        {selectedShipping ? (
          <div className="flex justify-between text-black">
            <span>
              Frete {selectedShipping.name}
              {selectedShipping.custom_delivery_time && (
                <span className="text-xs text-gray-500 block">
                  {selectedShipping.custom_delivery_time}
                </span>
              )}
            </span>
            <span>R$ {selectedShipping.price.toFixed(2)}</span>
          </div>
        ) : (
          <div className="flex justify-between text-black">
            <span>Frete</span>
            <span className="text-gray-500">A calcular</span>
          </div>
        )}
        
        {discount > 0 && (
          <div className="flex justify-between text-green-600">
            <span>Desconto</span>
            <span>- R$ {discount.toFixed(2)}</span>
          </div>
        )}
        <div className="border-t pt-2 mt-2">
          <div className="flex justify-between font-bold text-lg text-black">
            <span>Total</span>            
            <span>R$ {calculateTotal().toFixed(2)}</span>
          </div>
        </div>
      </div>

      {paymentMethod === 'pix' && currentStep < 4 && (
        <div className="mt-4 p-3 bg-green-50 rounded-lg border border-green-200">
          <div className="flex items-center text-green-700">
            <SiPix className="mr-2" />
            <span className="font-semibold">5% de desconto no PIX</span>
          </div>
          <p className="text-green-600 text-sm mt-1">
            Economize R$ {(calculateTotal() * 0.05).toFixed(2)}
          </p>
        </div>
      )}

      <div className="mt-6 pt-4 border-t border-gray-200">
        <p className="text-sm font-medium text-gray-700 mb-2">Navegação Rápida:</p>
        <div className="flex flex-wrap gap-2">
          <button
            onClick={() => goToStep(1)}
            className="text-xs bg-gray-100 hover:bg-gray-200 text-gray-700 px-3 py-1 rounded"
          >
            Dados Pessoais
          </button>
          <button
            onClick={() => goToStep(2)}
            className="text-xs bg-gray-100 hover:bg-gray-200 text-gray-700 px-3 py-1 rounded"
          >
            Entrega
          </button>
          <button
            onClick={() => goToStep(3)}
            className="text-xs bg-gray-100 hover:bg-gray-200 text-gray-700 px-3 py-1 rounded"
          >
            Pagamento
          </button>
          {currentStep === 4 && (
            <button
              onClick={() => goToStep(4)}
              className="text-xs bg-gray-100 hover:bg-gray-200 text-gray-700 px-3 py-1 rounded"
            >
              PIX
            </button>
          )}
        </div>
      </div>
    </div>
  );
};

// ✅ Main CheckoutPage Component
const CheckoutPage = () => {
  const location = useLocation();
  const navigate = useNavigate();
  
  // ✅ CART ITEMS: Load instantly from location state
  const [cartItems, setCartItems] = useState([]);
  const [cartTotal, setCartTotal] = useState(0);
  
  // ✅ USER DATA: Load asynchronously
  const [customerData, setCustomerData] = useState({
    name: '',
    email: '',
    phone: '',
    cpf: ''
  });
  const [shippingData, setShippingData] = useState({
    cep: '',
    street: '',
    number: '',
    complement: '',
    hood: '',
    city: '',
    state: ''
  });
  const [selectedShipping, setSelectedShipping] = useState(null);
  const [paymentMethod, setPaymentMethod] = useState('pix');
  const [cardData, setCardData] = useState({
    number: '',
    name: '',
    expiry: '',
    cvv: '',
    installments: 1
  });
  const [pixData, setPixData] = useState(null);
  const [loading, setLoading] = useState(false);
  const [currentStep, setCurrentStep] = useState(1);
  const [couponCode, setCouponCode] = useState('');
  const [discount, setDiscount] = useState(0);
  const [userAddresses, setUserAddresses] = useState([]);
  const [selectedAddressId, setSelectedAddressId] = useState(null);
  
  // ✅ LOADING STATES: Separate for different data types
  const [isProfileLoaded, setIsProfileLoaded] = useState(false);
  const [isAddressesLoaded, setIsAddressesLoaded] = useState(false);
  const [isUserDataLoading, setIsUserDataLoading] = useState(true);

  // ✅ 1. INSTANT: Load cart items from location state
  useEffect(() => {
    console.log("📍 Location state:", location.state);
    
    if (location.state?.selectedCartItems) {
      console.log("🛒 Using SELECTED items from cart:", location.state.selectedCartItems);
      setCartItems(location.state.selectedCartItems);
      setCartTotal(location.state.selectedItemsTotal || 0);
    } else {
      console.log("⚠️ No selected items found in location state");
      setCartItems([]);
    }
  }, [location.state]);

  // ✅ 2. ASYNC: Load user data (profile, addresses)
  useEffect(() => {
    let isMounted = true;

    const loadUserData = async () => {
      if (!isMounted) return;

      try {
        setIsUserDataLoading(true);
        console.log("🔄 Loading user data asynchronously...");
        
        // Get CSRF token first
        await CsrfTokenManager.getCsrfToken();
        
        // Load user profile and addresses in parallel
        await Promise.all([
          fetchUserProfile(),
          fetchUserAddresses()
        ]);
        
        console.log("✅ User data loaded successfully");
      } catch (error) {
        console.error("❌ Error loading user data:", error);
      } finally {
        if (isMounted) {
          setIsUserDataLoading(false);
        }
      }
    };

    loadUserData();

    return () => {
      isMounted = false;
    };
  }, []);

  // ✅ Helper function
  const isUsingSelectedItems = () => {
    return !!location.state?.selectedCartItems;
  };

  const saveOrderToBackend = async (orderData) => {
    try {
      console.log('🔄 Saving order to backend database...');
      
      const result = await OrderAPI.createOrder(orderData);
      
      if (result.success) {
        console.log('✅ Order saved to database:', result.data);
        return {
          success: true,
          orderId: result.data.order_id,
          orderNumber: result.data.order_number,
          data: result.data
        };
      } else {
        console.error('❌ Backend order creation failed:', result.message);
        return {
          success: false,
          error: result.message
        };
      }
    } catch (error) {
      console.error('❌ Error saving order to backend:', error);
      return {
        success: false,
        error: error.message
      };
    }
  };

  const generateMockPixCode = (amount) => {
    const randomId = Math.random().toString(36).substring(2, 15);
    const formattedAmount = amount.toFixed(2);
    
    return `00020126580014br.gov.bcb.pix0136${randomId}520400005303986540${formattedAmount}5802BR5913LOJA EXEMPLO6008SAO PAULO62070503***6304${Math.random().toString(36).substring(2, 6).toUpperCase()}`;
  };

  const processCheckout = async () => {
    if (cartItems.length === 0) {
      alert('Seu carrinho está vazio!');
      return;
    }
  
    const missingCustomer = [
      !customerData.name?.trim() && 'Nome',
      !customerData.email?.trim() && 'E-mail',
      !customerData.phone?.trim() && 'Telefone',
      !customerData.cpf?.trim() && 'CPF'
    ].filter(Boolean);
  
    if (missingCustomer.length > 0) {
      alert(`Preencha os campos obrigatórios: ${missingCustomer.join(', ')}`);
      return;
    }
  
    const missingAddress = [
      !shippingData.cep?.trim() && 'CEP',
      !shippingData.street?.trim() && 'Rua',
      !shippingData.number?.trim() && 'Número',
      !shippingData.hood?.trim() && 'Bairro',
      !shippingData.city?.trim() && 'Cidade',
      !shippingData.state?.trim() && 'Estado'
    ].filter(Boolean);
  
    if (missingAddress.length > 0) {
      alert(`Preencha o endereço completo: ${missingAddress.join(', ')}`);
      return;
    }
  
    if (!selectedShipping) {
      alert('Selecione uma opção de frete');
      return;
    }
  
    setLoading(true);
  
    try {
      const orderPayload = generateOrderJSON();
  
      console.log('%c ORDER PAYLOAD → /api/infos ', 'background:#006400;color:#fff;font-weight:bold;', orderPayload);
  
      const backendResult = await saveOrderToBackend(orderPayload);
  
      if (!backendResult.success) {
        throw new Error(backendResult.message || 'Falha ao salvar o pedido no servidor');
      }
  
      console.log('Order created in database:', backendResult.data);
  
      const orderId = backendResult.data.order_id;
      const orderNumber = backendResult.data.order_number;
  
      if (paymentMethod === 'pix') {
        const mockPixCode = generateMockPixCode(calculateTotal());
        const mockQrCodeUrl = `https://api.qrserver.com/v1/create-qr-code/?size=300x300&data=${encodeURIComponent(mockPixCode)}`;
  
        setPixData({
          pixQrCode: mockQrCodeUrl,
          pixCode: mockPixCode,
          amount: calculateTotal(),
          backendOrderId: orderId,
          backendOrderNumber: orderNumber
        });
  
        await OrderAPI.updateOrderStatus(orderId, 'pending', 'pending');
        setCurrentStep(4);
  
      } else {
        await OrderAPI.updateOrderStatus(orderId, 'processing', 'paid');
        alert(`Pedido #${orderNumber} criado com sucesso!`);
        await clearCart();
        window.location.href = '/';
      }
  
    } catch (error) {
      console.error('Checkout error:', error);
      const msg = error.message.includes('HTTP 500')
        ? 'Erro no servidor. Verifique os dados e tente novamente.'
        : error.message;
      alert('Erro no checkout: ' + msg);
    } finally {
      setLoading(false);
    }
  };

  const clearCart = async () => {
    try {
      const csrfToken = await CsrfTokenManager.getCsrfToken();
      
      const response = await fetch('http://localhost:8000/api/cart/clear', {
        method: 'POST',
        credentials: 'include',
        headers: {
          'Accept': 'application/json',
          'X-Requested-With': 'XMLHttpRequest',
          'X-XSRF-TOKEN': csrfToken,
        },
      });
      
      if (response.ok) {
        console.log('🛒 Cart cleared after successful order');
      }
    } catch (error) {
      console.error('Error clearing cart:', error);
    }
  };

  const calculateSubtotal = () => {
    if (cartTotal > 0) {
      return cartTotal;
    }
    
    return cartItems.reduce((total, item) => {
      const price = item.product?.price || item.price || 0;
      const quantity = item.quantity || 1;
      return total + (price * quantity);
    }, 0);
  };

  const calculateShipping = () => {
    return selectedShipping ? selectedShipping.price : 0;
  };

  const calculateTotal = () => {
    const subtotal = calculateSubtotal();
    const shipping = calculateShipping();
    const total = subtotal + shipping - discount;
    
    const discountedTotal = paymentMethod === 'pix' ? total * 0.95 : total;
    
    return discountedTotal;
  };
  
  const generateOrderJSON = () => {
    const subtotal      = calculateSubtotal();
    const shippingCost  = calculateShipping();
    const rawTotal      = subtotal + shippingCost - discount;
    const total         = paymentMethod === 'pix' ? rawTotal * 0.95 : rawTotal;
    const discountAmt   = paymentMethod === 'pix' ? rawTotal * 0.05 : discount;

    const items = cartItems.map(item => {
      const product = item.product || {};
      const price   = Number(product.price || item.price || 0);
      const qty     = Number(item.quantity || 1);

      return {
        product_id:   item.product_id || product.id || null,
        product_name: product.title || product.name || `Produto ${item.product_id || ''}`,
        quantity:     qty,
        price:        price,
        total:        price * qty
      };
    });

    const payload = {
      customer_name:  (customerData.name   ?? '').toString().trim(),
      customer_email: (customerData.email  ?? '').toString().trim(),
      customer_phone: (customerData.phone  ?? '').toString().trim(),
      customer_cpf:   (customerData.cpf    ?? '').toString().trim(),

      subtotal:      Number(subtotal.toFixed(2)),
      shipping:      Number(shippingCost.toFixed(2)),
      discount:      Number(discountAmt.toFixed(2)),
      total:         Number(total.toFixed(2)),

      payment_method: paymentMethod,
      payment_status: 'pending',
      status:         'pending',

      shipping_street:        (shippingData.street        ?? '').toString().trim(),
      shipping_number:        (shippingData.number        ?? '').toString().trim(),
      shipping_complement:    (shippingData.complement    ?? null),
      shipping_neighborhood:  (shippingData.hood          ?? '').toString().trim(),
      shipping_city:          (shippingData.city          ?? '').toString().trim(),
      shipping_state:         (shippingData.state         ?? '').toString().trim(),
      shipping_zip_code:      (shippingData.cep?.replace(/\D/g, '') ?? '').toString(),

      items: items,

      metadata: {
        is_mock: true,
        coupon_applied: discount > 0 ? couponCode : null,
        platform: 'web',
        timestamp: new Date().toISOString(),
        selected_items_only: isUsingSelectedItems()
      }
    };

    console.log('%c ORDER PAYLOAD → /api/infos ', 'background:#006400;color:#fff;', payload);
    return payload;
  };

  const fetchUserProfile = async () => {
    try {
      console.log("🔄 Fetching user profile...");

      const res = await fetch("http://localhost:8000/profile", { 
        method: "GET",
        credentials: "include",
        headers: {
          "Accept": "application/json",
          "X-Requested-With": "XMLHttpRequest",
        },
      });

      if (res.ok) {
        const userData = await res.json();
        console.log("✅ USER PROFILE LOADED:", userData);
        
        const updatedCustomerData = {
          name: userData.name || '',
          email: userData.email || '',
          phone: userData.tel || userData.phone || userData.telefone || '',
          cpf: userData.cpf || userData.document || ''
        };
        
        setCustomerData(updatedCustomerData);
        setIsProfileLoaded(true);
        
      } else {
        console.log("⚠️ Could not fetch user profile, using empty form");
        setIsProfileLoaded(true);
      }
    } catch (err) {
      console.error("❌ Error fetching user profile:", err.message);
      setIsProfileLoaded(true);
    }
  };

  const fetchUserAddresses = async () => {
    try {
      console.log("🔄 Fetching user addresses...");

      const possibleEndpoints = [
        'http://localhost:8000/addresses',
        'http://localhost:8000/api/address'
      ];

      let addresses = [];

      for (const endpoint of possibleEndpoints) {
        try {
          console.log(`🔍 Trying endpoint: ${endpoint}`);
          const res = await fetch(endpoint, {
            method: "GET",
            credentials: "include",
            headers: {
              "Accept": "application/json",
              "X-Requested-With": "XMLHttpRequest",
            },
          });

          if (res.ok) {
            const data = await res.json();
            console.log(`✅ ADDRESSES LOADED from ${endpoint}:`, data);
            
            if (Array.isArray(data)) {
              addresses = data;
            } else if (data.data && Array.isArray(data.data)) {
              addresses = data.data;
            } else if (data.addresses && Array.isArray(data.addresses)) {
              addresses = data.addresses;
            } else {
              addresses = [];
            }
            
            break;
          }
        } catch (err) {
          console.log(`❌ Endpoint ${endpoint} error:`, err.message);
          continue;
        }
      }

      if (addresses.length === 0) {
        console.log("⚠️ No addresses found from APIs");
        addresses = [];
      }

      setUserAddresses(addresses);
      
      if (addresses.length > 0) {
        const primaryAddress = addresses.find(addr => addr.is_primary) || addresses[0];
        setSelectedAddressId(primaryAddress.id);
        
        setShippingData({
          cep: primaryAddress.zip || primaryAddress.cep || primaryAddress.postal_code || '',
          street: primaryAddress.street || primaryAddress.logradouro || primaryAddress.address || '',
          number: primaryAddress.number || primaryAddress.numero || '',
          complement: primaryAddress.complement || primaryAddress.complemento || '',
          hood: primaryAddress.hood || primaryAddress.bairro || '',
          city: primaryAddress.city || primaryAddress.cidade || '',
          state: primaryAddress.state || primaryAddress.estado || primaryAddress.uf || ''
        });
      }
      
    } catch (err) {
      console.error("❌ Error fetching addresses:", err.message);
      setUserAddresses([]);
    } finally {
      setIsAddressesLoaded(true);
    }
  };

  const handleAddressSelect = (addressId) => {
    setSelectedAddressId(addressId);
    const selectedAddress = userAddresses.find(addr => addr.id === addressId);
    if (selectedAddress) {
      setShippingData({
        cep: selectedAddress.zip || selectedAddress.cep || selectedAddress.postal_code || '',
        street: selectedAddress.street || selectedAddress.logradouro || selectedAddress.address || '',
        number: selectedAddress.number || selectedAddress.numero || '',
        complement: selectedAddress.complement || selectedAddress.complemento || '',
        hood: selectedAddress.hood || selectedAddress.bairro || '',
        city: selectedAddress.city || selectedAddress.cidade || '',
        state: selectedAddress.state || selectedAddress.estado || selectedAddress.uf || ''
      });
    }
  };

  const handleShippingSelect = (shipping) => {
    setSelectedShipping(shipping);
  };

  const goToStep = (step) => {
    setCurrentStep(step);
  };

  const nextStep = () => {
    setCurrentStep(prev => prev + 1);
  };

  const prevStep = () => {
    setCurrentStep(prev => prev - 1);
  };

  const applyCoupon = () => {
    if (couponCode === 'DESCONTO10') {
      setDiscount(calculateSubtotal() * 0.1);
      alert('Cupom aplicado! 10% de desconto.');
    } else {
      alert('Cupom inválido');
    }
  };

  const copyPixCode = () => {
    if (pixData) {
      navigator.clipboard.writeText(pixData.pixCode);
      alert('Código PIX copiado!');
    }
  };

  // ✅ Only show loading for user data, cart loads instantly
  if (cartItems.length === 0) {
    return (
      <div className="min-h-screen bg-gray-50 pt-20 flex items-center justify-center" style={{ fontFamily: 'Calibri, sans-serif' }}>
        <div className="text-center bg-white p-8 rounded-lg shadow-md max-w-md w-full mx-4">
          <div className="w-20 h-20 bg-gray-100 rounded-full flex items-center justify-center mx-auto mb-4">
            <FiShoppingCart size={32} className="text-gray-400" />
          </div>
          <h2 className="text-2xl font-bold text-gray-900 mb-2">
            {isUsingSelectedItems() ? 'Itens Não Encontrados' : 'Carrinho Vazio'}
          </h2>
          <p className="text-gray-600 mb-6">
            {isUsingSelectedItems() 
              ? "Os itens selecionados não estão mais disponíveis. Por favor, volte ao carrinho e selecione os itens novamente."
              : "Seu carrinho está vazio. Adicione produtos antes de finalizar a compra."
            }
          </p>
          <button
            onClick={() => navigate('/cart')}
            className="bg-green-600 hover:bg-green-700 text-white py-3 px-6 rounded-lg font-semibold transition-colors"
          >
            Voltar para Carrinho
          </button>
        </div>
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-gray-50 pt-20" style={{ fontFamily: 'Calibri, sans-serif' }}>
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 py-8">
        <StepIndicator currentStep={currentStep} goToStep={goToStep} />
        
        <div className="grid grid-cols-1 lg:grid-cols-3 gap-8">
          <div className="lg:col-span-2 space-y-6">
            {currentStep === 1 && (
              <CustomerStep 
                customerData={customerData}
                setCustomerData={setCustomerData}
                isProfileLoaded={isProfileLoaded}
                fetchUserProfile={fetchUserProfile}
                nextStep={nextStep}
              />
            )}
            {currentStep === 2 && (
              <ShippingStep 
                shippingData={shippingData}
                setShippingData={setShippingData}
                userAddresses={userAddresses}
                selectedAddressId={selectedAddressId}
                handleAddressSelect={handleAddressSelect}
                fetchUserAddresses={fetchUserAddresses}
                selectedShipping={selectedShipping}
                onShippingSelect={handleShippingSelect}
                cartItems={cartItems}
                prevStep={prevStep}
                nextStep={nextStep}
              />
            )}
            {currentStep === 3 && (
              <PaymentStep 
                paymentMethod={paymentMethod}
                setPaymentMethod={setPaymentMethod}
                cardData={cardData}
                setCardData={setCardData}
                calculateTotal={calculateTotal}
                prevStep={prevStep}
                processCheckout={processCheckout}
                loading={loading}
              />
            )}
            {currentStep === 4 && (
              <PixStep 
                pixData={pixData}
                copyPixCode={copyPixCode}
                goToStep={goToStep}
              />
            )}
          </div>

          <div className="lg:col-span-1">
            <OrderSummary 
              cartItems={cartItems}
              couponCode={couponCode}
              setCouponCode={setCouponCode}
              applyCoupon={applyCoupon}
              calculateSubtotal={calculateSubtotal}
              calculateShipping={calculateShipping}
              discount={discount}
              calculateTotal={calculateTotal}
              paymentMethod={paymentMethod}

              isUsingSelectedItems={isUsingSelectedItems()}
              navigate={navigate}
            />
          </div>
        </div>
      </div>
    </div>
  );
};
