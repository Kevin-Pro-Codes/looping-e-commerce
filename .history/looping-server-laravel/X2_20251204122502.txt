class Info extends Model
{

    protected $table = 'infos';

    protected $fillable = [
        'order_number',
        'customer_name',
        'customer_email',
        'customer_phone',
        'customer_cpf',
        'total',
        'subtotal',
        'shipping',
        'discount',
        'payment_method',
        'payment_status',
        'stock_updated',           // ✅ ADDED
        'stock_updated_at',        // ✅ ADDED
        'status',
        'shipping_street',
        'shipping_number',
        'shipping_complement',
        'shipping_neighborhood',
        'shipping_city',
        'shipping_state',
        'shipping_zip_code',
        'items',       
        'metadata'
    ];

    protected $casts = [
        'total' => 'decimal:2',
        'subtotal' => 'decimal:2',
        'shipping' => 'decimal:2',
        'discount' => 'decimal:2',
        'stock_updated' => 'boolean',  // ✅ ADDED
        'stock_updated_at' => 'datetime',  // ✅ ADDED
        'items' => 'array',
        'metadata' => 'array',
        'created_at' => 'datetime',
        'updated_at' => 'datetime',
       
    ];

    /**
     * Boot method to handle JSON encoding issues
     */
    protected static function boot()
    {
        parent::boot();

        static::retrieved(function ($info) {
            // Ensure JSON fields are properly handled
            try {
                if ($info->items && is_string($info->items)) {
                    $info->items = json_decode($info->items, true);
                }
            
                if ($info->metadata && is_string($info->metadata)) {
                    $info->metadata = json_decode($info->metadata, true);
                }
            } catch (\Exception $e) {
                Log::error('Error decoding JSON fields for info ' . $info->id . ': ' . $e->getMessage());
            }
        });

     
    }

}

    /**
     * Update info status and payment status WITH STOCK SYNC
     * PUT /api/infos/{id}/status
     */
    public function updateStatus($id, Request $request, StockService $stockService): JsonResponse
    {
        try {
            Log::info('=== INFO STATUS UPDATE START ===', ['info_id' => $id]);

            // Validate input
            $validated = $request->validate([
                'status' => 'required|string|in:pending,processing,shipped,delivered,cancelled',
                'payment_status' => 'required|string|in:pending,paid,failed,refunded'
            ]);

            // Find info
            $info = Info::find($id);
            if (!$info) {
                return response()->json([
                    'success' => false,
                    'message' => 'Info not found',
                    'debug' => ['info_id' => $id]
                ], 404);
            }

            $oldPaymentStatus = $info->payment_status;
            $oldStatus = $info->status;

            // Update fields
            $info->status = $validated['status'];
            $info->payment_status = $validated['payment_status'];
            
            // ✅ HANDLE STOCK SYNC BASED ON STATUS CHANGES
            if (($oldPaymentStatus === 'paid' && $validated['payment_status'] === 'refunded') ||
                ($oldStatus !== 'cancelled' && $validated['status'] === 'cancelled')) {
                // Restore stock for refunds or cancellations
                $stockService->restoreStockForOrder($info);
                Log::info('Stock restored for order due to refund/cancellation', [
                    'info_id' => $info->id,
                    'old_payment_status' => $oldPaymentStatus,
                    'new_payment_status' => $validated['payment_status'],
                    'old_status' => $oldStatus,
                    'new_status' => $validated['status']
                ]);
            } elseif ($oldPaymentStatus !== 'paid' && $validated['payment_status'] === 'paid') {
                // Deduct stock when payment is confirmed
                $stockUpdated = $stockService->deductStockFromOrder($info);
                if (!$stockUpdated) {
                    return response()->json([
                        'success' => false,
                        'message' => 'Failed to update stock. Insufficient stock available.',
                        'debug' => ['info_id' => $info->id]
                    ], 400);
                }
                Log::info('Stock deducted for order due to payment confirmation', [
                    'info_id' => $info->id,
                    'old_payment_status' => $oldPaymentStatus,
                    'new_payment_status' => $validated['payment_status']
                ]);
            }
            
            $info->save();

            Log::info('Info status updated successfully', [
                'info_id' => $info->id,
                'status' => $info->status,
                'payment_status' => $info->payment_status,
                'stock_updated' => $info->stock_updated,
                'stock_updated_at' => $info->stock_updated_at
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Info status updated successfully',
                'data' => [
                    'info_id' => $info->id,
                    'order_number' => $info->order_number,
                    'status' => $info->status,
                    'payment_status' => $info->payment_status,
                    'stock_updated' => $info->stock_updated,
                    'stock_updated_at' => $info->stock_updated_at
                ],
                'debug' => [
                    'timestamp' => now()->format('c')
                ]
            ]);

        } catch (Exception $e) {
            Log::error('=== INFO STATUS UPDATE ERROR ===');
            Log::error('Error: ' . $e->getMessage());
            Log::error('File: ' . $e->getFile() . ':' . $e->getLine());
            Log::error('Request data: ', $request->all());

            return response()->json([
                'success' => false,
                'message' => 'Failed to update info status',
                'error' => $e->getMessage(),
                'debug' => [
                    'exception' => get_class($e),
                    'message' => $e->getMessage(),
                    'timestamp' => now()->format('c')
                ]
            ], 500);
        }
    }

    /**
     * Store a newly created info (POST /api/infos) WITH STOCK CHECK OPTION
     */
    public function store(Request $request, StockService $stockService): JsonResponse
    {
        try {
            Log::info('=== INFO STORE START ===');
            Log::info('Info data received:', $request->all());

            // More flexible validation with proper defaults
            $validated = $request->validate([
                'customer_name' => 'required|string|max:255',
                'customer_email' => 'required|email',
                'customer_phone' => 'required|string',
                'customer_cpf' => 'required|string',
                'total' => 'required|numeric|min:0',
                'subtotal' => 'required|numeric|min:0',
                'shipping' => 'required|numeric|min:0',
                'discount' => 'sometimes|numeric|min:0',
                'payment_method' => 'required|string|in:pix,credit,debit,boleto',
                'payment_status' => 'required|string|in:pending,paid,failed,refunded',
                'status' => 'required|string|in:pending,processing,shipped,delivered,cancelled',
                'shipping_street' => 'required|string',
                'shipping_number' => 'required|string',
                'shipping_complement' => 'nullable|string',
                'shipping_neighborhood' => 'required|string',
                'shipping_city' => 'required|string',
                'shipping_state' => 'required|string|size:2',
                'shipping_zip_code' => 'required|string',
                'items' => 'required|array|min:1',
                'items.*.product_name' => 'required|string',
                'items.*.quantity' => 'required|integer|min:1',
                'items.*.price' => 'required|numeric|min:0',
                'items.*.total' => 'required|numeric|min:0',
                // ✅ ADDED: Validation for image, color, and size
                'items.*.image' => 'sometimes|string|nullable',
                'items.*.color' => 'sometimes|string|nullable',
                'items.*.size' => 'sometimes|string|nullable',
            ]);

            // ✅ OPTIONAL: Check stock before creating order
            if ($request->has('check_stock') && $request->check_stock == true) {
                $stockCheck = $stockService->checkStockAvailability($validated['items']);
                $allAvailable = collect($stockCheck)->every(function($result) {
                    return $result['available'] === true;
                });
                
                if (!$allAvailable) {
                    return response()->json([
                        'success' => false,
                        'message' => 'Some items are out of stock',
                        'stock_check' => $stockCheck
                    ], 400);
                }
            }

            // Generate order number if not provided
            $orderNumber = $request->order_number ?? 'ORD-' . date('Ymd') . '-' . strtoupper(substr(uniqid(), -6));

            // Prepare the data with defaults
            $infoData = [
                'order_number' => $orderNumber,
                'customer_name' => $validated['customer_name'],
                'customer_email' => $validated['customer_email'],
                'customer_phone' => $validated['customer_phone'],
                'customer_cpf' => $validated['customer_cpf'],
                'total' => $validated['total'],
                'subtotal' => $validated['subtotal'],
                'shipping' => $validated['shipping'],
                'discount' => $validated['discount'] ?? 0,
                'payment_method' => $validated['payment_method'],
                'payment_status' => $validated['payment_status'] ?? 'pending',
                'stock_updated' => false,  // ✅ ADDED - default false
                'stock_updated_at' => null, // ✅ ADDED - default null
                'status' => $validated['status'] ?? 'pending',
                'shipping_street' => $validated['shipping_street'],
                'shipping_number' => $validated['shipping_number'],
                'shipping_complement' => $validated['shipping_complement'] ?? null,
                'shipping_neighborhood' => $validated['shipping_neighborhood'],
                'shipping_city' => $validated['shipping_city'],
                'shipping_state' => $validated['shipping_state'],
                'shipping_zip_code' => $validated['shipping_zip_code'],
                'items' => json_encode($validated['items']),
                'metadata' => $request->metadata ? json_encode($request->metadata) : null,
            ];

            Log::info('Creating info with data:', $infoData);

            $info = Info::create($infoData);

            // ✅ If payment is already marked as paid, deduct stock immediately
            if ($validated['payment_status'] === 'paid') {
                $stockUpdated = $stockService->deductStockFromOrder($info);
                if (!$stockUpdated) {
                    // If stock deduction fails, delete the order
                    $info->delete();
                    return response()->json([
                        'success' => false,
                        'message' => 'Failed to create order. Insufficient stock available.'
                    ], 400);
                }
                Log::info('Stock deducted immediately for paid order', [
                    'info_id' => $info->id,
                    'order_number' => $info->order_number
                ]);
            }

            Log::info('Info created successfully:', [
                'info_id' => $info->id, 
                'order_number' => $info->order_number,
                'stock_updated' => $info->stock_updated
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Order created successfully',
                'data' => [
                    'order_id' => $info->id,
                    'order_number' => $info->order_number,
                    'order' => $info,
                    'stock_updated' => $info->stock_updated
                ],
                'debug' => [
                    'timestamp' => now()->format('c'),
                    'info_id' => $info->id
                ]
            ], 201);

        } catch (\Illuminate\Validation\ValidationException $e) {
            Log::error('=== INFO STORE VALIDATION ERROR ===');
            Log::error('Validation errors:', $e->errors());
            Log::error('Request data: ', $request->all());

            return response()->json([
                'success' => false,
                'message' => 'Validation failed',
                'errors' => $e->errors(),
                'debug' => [
                    'timestamp' => now()->format('c')
                ]
            ], 422);

        } catch (Exception $e) {
            Log::error('=== INFO STORE ERROR ===');
            Log::error('Error: ' . $e->getMessage());
            Log::error('File: ' . $e->getFile() . ':' . $e->getLine());
            Log::error('Request data: ', $request->all());

            return response()->json([
                'success' => false,
                'message' => 'Failed to create order',
                'error' => $e->getMessage(),
                'debug' => [
                    'exception' => get_class($e),
                    'message' => $e->getMessage(),
                    'timestamp' => now()->format('c')
                ]
            ], 500);
        }
    }

    /**
     * ✅ NEW: Confirm payment and update stock
     * POST /api/infos/{id}/confirm-payment
     */
    public function confirmPayment($id, Request $request, StockService $stockService): JsonResponse
    {
        try {
            Log::info('=== CONFIRM PAYMENT START ===', ['info_id' => $id]);

            // Find the order
            $info = Info::find($id);
            if (!$info) {
                return response()->json([
                    'success' => false,
                    'message' => 'Order not found'
                ], 404);
            }

            // Check if payment is already confirmed
            if ($info->payment_status === 'paid' && $info->stock_updated) {
                return response()->json([
                    'success' => true,
                    'message' => 'Payment already confirmed and stock updated',
                    'data' => [
                        'order_id' => $info->id,
                        'order_number' => $info->order_number,
                        'payment_status' => $info->payment_status,
                        'stock_updated' => $info->stock_updated
                    ]
                ]);
            }

            // Update payment status
            $info->payment_status = 'paid';
            $info->status = 'processing';
            
            // Deduct stock
            $stockUpdated = $stockService->deductStockFromOrder($info);
            
            if (!$stockUpdated) {
                return response()->json([
                    'success' => false,
                    'message' => 'Failed to update stock. Please check stock availability.'
                ], 400);
            }

            $info->save();

            Log::info('Payment confirmed and stock updated', [
                'info_id' => $info->id,
                'order_number' => $info->order_number,
                'stock_updated' => $info->stock_updated,
                'stock_updated_at' => $info->stock_updated_at
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Payment confirmed and stock updated successfully',
                'data' => [
                    'order_id' => $info->id,
                    'order_number' => $info->order_number,
                    'payment_status' => $info->payment_status,
                    'stock_updated' => $info->stock_updated,
                    'stock_updated_at' => $info->stock_updated_at
                ]
            ]);

        } catch (Exception $e) {
            Log::error('Confirm payment error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Failed to confirm payment',
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * ✅ NEW: Check stock availability before placing order
     * POST /api/infos/check-stock
     */
    public function checkStock(Request $request, StockService $stockService): JsonResponse
    {
        try {
            $validated = $request->validate([
                'items' => 'required|array|min:1',
                'items.*.product_name' => 'required|string',
                'items.*.quantity' => 'required|integer|min:1',
                'items.*.color' => 'sometimes|string|nullable',
                'items.*.size' => 'sometimes|string|nullable',
            ]);

            $stockCheck = $stockService->checkStockAvailability($validated['items']);
            
            $allAvailable = true;
            $messages = [];
            $unavailableItems = [];
            
            foreach ($stockCheck as $key => $result) {
                if (!$result['available']) {
                    $allAvailable = false;
                    $item = $validated['items'][$key];
                    $messages[] = "Estoque insuficiente para {$item['product_name']}: Disponível {$result['available_stock']}, Solicitado {$item['quantity']}";
                    $unavailableItems[] = [
                        'product_name' => $item['product_name'],
                        'requested_quantity' => $item['quantity'],
                        'available_stock' => $result['available_stock'],
                        'color' => $item['color'] ?? null,
                        'size' => $item['size'] ?? null
                    ];
                }
            }

            return response()->json([
                'success' => true,
                'all_available' => $allAvailable,
                'stock_check' => $stockCheck,
                'messages' => $messages,
                'unavailable_items' => $unavailableItems,
                'debug' => [
                    'items_count' => count($validated['items']),
                    'timestamp' => now()->format('c')
                ]
            ]);

        } catch (Exception $e) {
            Log::error('Check stock error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Failed to check stock',
                'error' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * ✅ NEW: Manual stock sync for an order
     * POST /api/infos/{id}/sync-stock
     */
    public function syncStock($id, StockService $stockService): JsonResponse
    {
        try {
            Log::info('=== MANUAL STOCK SYNC START ===', ['info_id' => $id]);

            $info = Info::find($id);
            if (!$info) {
                return response()->json([
                    'success' => false,
                    'message' => 'Order not found'
                ], 404);
            }

            // Check if order is eligible for stock deduction
            if (!$info->canDeductStock()) {
                return response()->json([
                    'success' => false,
                    'message' => 'Order is not eligible for stock deduction',
                    'details' => [
                        'payment_status' => $info->payment_status,
                        'status' => $info->status,
                        'stock_updated' => $info->stock_updated
                    ]
                ], 400);
            }

            $stockUpdated = $stockService->deductStockFromOrder($info);
            
            if (!$stockUpdated) {
                return response()->json([
                    'success' => false,
                    'message' => 'Failed to sync stock'
                ], 400);
            }

            return response()->json([
                'success' => true,
                'message' => 'Stock synced successfully',
                'data' => [
                    'order_id' => $info->id,
                    'order_number' => $info->order_number,
                    'stock_updated' => $info->stock_updated,
                    'stock_updated_at' => $info->stock_updated_at
                ]
            ]);

        } catch (Exception $e) {
            Log::error('Manual stock sync error: ' . $e->getMessage());
            return response()->json([
                'success' => false,
                'message' => 'Failed to sync stock',
                'error' => $e->getMessage()
            ], 500);
        }
    }

use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\Schema;

class Stock extends Model
{
    use HasFactory;

    protected $table = 'stock';
    
    protected $fillable = [
        'product_id', 'name', 'sku', 'category', 'supplier',
        'cost', 'profit', 'sale_price', 'total_value',
        'color', 'size', 'stock', 'last_updated'
    ];

    protected $casts = [
        'cost' => 'decimal:2',
        'profit' => 'decimal:2',
        'sale_price' => 'decimal:2',
        'total_value' => 'decimal:2',
        'stock' => 'integer',
        'last_updated' => 'date'
    ];

    public static function getGroupedStock()
    {
        if (!Schema::hasTable('stock')) {
            return collect();
        }

        return self::orderBy('product_id')
            ->orderBy('color')
            ->orderBy('size')
            ->get()
            ->groupBy('product_id')
            ->map(function ($items, $productId) {
                $firstItem = $items->first();
                
                return [
                    'id' => $firstItem->id,
                    'product_id' => $productId,
                    'name' => $firstItem->name,
                    'category' => $firstItem->category,
                    'supplier' => $firstItem->supplier,
                    'cost' => (float) $firstItem->cost,
                    'profit' => (float) $firstItem->profit,
                    'sale_price' => (float) $firstItem->sale_price,
                    'total_value' => (float) $firstItem->total_value,
                    'last_updated' => $firstItem->last_updated,
                    'created_at' => $firstItem->created_at?->format('Y-m-d H:i:s'),
                    'updated_at' => $firstItem->updated_at?->format('Y-m-d H:i:s'),
                    'variants' => $items->map(function ($item) {
                        return [
                            'id' => $item->id,
                            'color' => $item->color,
                            'size' => $item->size,
                            'stock' => $item->stock,
                            'sku' => $item->sku,
                        ];
                    })->values()->toArray()
                ];
            })->values();
    }

    public static function getLowStock()
    {
        if (!Schema::hasTable('stock')) {
            return collect();
        }

        return self::where('stock', '>', 0)
            ->where('stock', '<=', 5)
            ->get();
    }

    public static function getOutOfStock()
    {
        if (!Schema::hasTable('stock')) {
            return collect();
        }

        return self::where('stock', 0)->get();
    }
}

<?php

namespace App\Services;

use App\Models\Stock;
use App\Models\Info;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class StockService
{
    /**
     * Deduct stock for an order
     */
    public function deductStockFromOrder(Info $order): bool
    {
        try {
            // If stock already updated, return true
            if ($order->stock_updated) {
                Log::info('Stock already deducted for order:', ['order_id' => $order->id]);
                return true;
            }

            // Only deduct stock for paid orders
            if ($order->payment_status !== 'paid') {
                Log::warning('Cannot deduct stock for unpaid order:', [
                    'order_id' => $order->id,
                    'payment_status' => $order->payment_status
                ]);
                return false;
            }

            $items = $order->items;
            if (empty($items)) {
                Log::warning('No items in order:', ['order_id' => $order->id]);
                return false;
            }

            DB::beginTransaction();

            foreach ($items as $item) {
                $this->deductItemStock($item);
            }

            // Mark the order as having stock deducted
            $order->stock_updated = true;
            $order->stock_updated_at = now();
            $order->save();

            DB::commit();

            Log::info('Stock deducted successfully for order:', [
                'order_id' => $order->id,
                'order_number' => $order->order_number,
                'items_count' => count($items)
            ]);

            return true;

        } catch (\Exception $e) {
            DB::rollBack();
            Log::error('Failed to deduct stock for order:', [
                'order_id' => $order->id,
                'error' => $e->getMessage(),
                'trace' => $e->getTraceAsString()
            ]);
            return false;
        }
    }

    /**
     * Deduct stock for a single item
     */
    private function deductItemStock(array $item): void
    {
        // Get product details from the item
        $productName = $item['product_name'] ?? null;
        $quantity = $item['quantity'] ?? 0;
        $color = $item['color'] ?? null;
        $size = $item['size'] ?? null;

        if (!$productName || $quantity <= 0) {
            Log::warning('Invalid item data for stock deduction:', $item);
            return;
        }

        // Find the stock record based on product name, color, and size
        $stock = Stock::where('name', $productName)
            ->when($color && $color !== 'Não especificada', function($query) use ($color) {
                return $query->where('color', $color);
            })
            ->when($size && $size !== 'Não especificado', function($query) use ($size) {
                return $query->where('size', $size);
            })
            ->first();

        if (!$stock) {
            // Try to find without color/size if they're null or unspecified
            $stock = Stock::where('name', $productName)->first();
            
            if (!$stock) {
                Log::warning('Stock record not found for product:', [
                    'product_name' => $productName,
                    'color' => $color,
                    'size' => $size
                ]);
                throw new \Exception("Stock record not found for product: {$productName}");
            }
        }

        $availableStock = $stock->stock - $stock->reserved;
        
        if ($availableStock < $quantity) {
            throw new \Exception("Insufficient stock for product: {$productName}. Available: {$availableStock}, Requested: {$quantity}");
        }

        // Update the stock
        $stock->decrement('stock', $quantity);
        $stock->last_updated = now()->format('Y-m-d');
        $stock->save();

        Log::info('Stock deducted for item:', [
            'product_name' => $productName,
            'quantity' => $quantity,
            'color' => $color,
            'size' => $size,
            'stock_id' => $stock->id
        ]);
    }
    
    /**
     * Restore stock (for cancelled orders or refunds)
     */
    public function restoreStockForOrder(Info $order): bool
    {
        try {
            // Only restore stock if it was previously deducted
            if (!$order->stock_updated) {
                return true;
            }

            $items = $order->items;
            if (empty($items)) {
                return false;
            }

            DB::beginTransaction();

            foreach ($items as $item) {
                $this->restoreItemStock($item);
            }

            // Mark the order as having stock restored
            $order->stock_updated = false;
            $order->stock_updated_at = null;
            $order->save();

            DB::commit();

            Log::info('Stock restored for order:', [
                'order_id' => $order->id,
                'order_number' => $order->order_number
            ]);

            return true;

        } catch (\Exception $e) {
            DB::rollBack();
            Log::error('Failed to restore stock for order:', [
                'order_id' => $order->id,
                'error' => $e->getMessage()
            ]);
            return false;
        }
    }

    /**
     * Restore stock for a single item
     */
    private function restoreItemStock(array $item): void
    {
        $productName = $item['product_name'] ?? null;
        $quantity = $item['quantity'] ?? 0;
        $color = $item['color'] ?? null;
        $size = $item['size'] ?? null;

        if (!$productName || $quantity <= 0) {
            return;
        }

        $stock = Stock::where('name', $productName)
            ->when($color && $color !== 'Não especificada', function($query) use ($color) {
                return $query->where('color', $color);
            })
            ->when($size && $size !== 'Não especificado', function($query) use ($size) {
                return $query->where('size', $size);
            })
            ->first();

        if (!$stock) {
            $stock = Stock::where('name', $productName)->first();
            
            if (!$stock) {
                return;
            }
        }

        // Restore the stock
        $stock->increment('stock', $quantity);
        $stock->last_updated = now()->format('Y-m-d');
        $stock->save();

        Log::info('Stock restored for item:', [
            'product_name' => $productName,
            'quantity' => $quantity,
            'stock_id' => $stock->id
        ]);
    }
