  // Skeleton loading component
  const SkeletonRow = () => (

        <div className="w-20 h-6 bg-gray-200 rounded"></div>
      </td>
      <td className="px-4 sm:px-6 py-4 whitespace-nowrap">
        <div className="flex space-x-2">
          <div className="w-6 h-6 bg-gray-200 rounded"></div>
          <div className="w-6 h-6 bg-gray-200 rounded"></div>
          <div className="w-6 h-6 bg-gray-200 rounded"></div>
        </div>
      </td>
    </tr>
  );

  // Helper function to extract item attributes
  const getItemAttributes = (item) => {
    const size = 
      item.size ||
      item.selectedSize || 
      item.tamanho || 
      item.variant_size ||
      item.product_variant?.size ||
      item.variant?.size ||
      item.product?.size ||
      (item.options && (item.options.size || item.options.tamanho)) ||
      item.attributes?.size ||
      item.metadata?.size ||
      'Não especificado';
    
    const color = 
      item.color ||
      item.selectedColor || 
      item.cor || 
      item.variant_color ||
      item.product_variant?.color ||
      item.variant?.color ||
      item.product?.color ||
      (item.options && (item.options.color || item.options.cor)) ||
      item.attributes?.color ||
      item.metadata?.color ||
      'Não especificada';
    
    // Get product image from various possible locations
    const image = 
      item.image ||
      item.product_image ||
      item.product_variant?.image ||
      item.variant?.image ||
      item.product?.image ||
      item.thumbnail ||
      item.main_image ||
      null;
    
    return { size, color, image };
  };

  // Helper function to get color name from hex code or color string
  const getColorName = (color) => {
    if (!color || color === 'Não especificada' || color === 'undefined') {
      return 'Não especificada';
    }
    
    const colorMap = {
      "#000000": "Preto",
      "#000": "Preto",
      "#FFFFFF": "Branco", 
      "#fff": "Branco",
      "#1E90FF": "Azul",
      "#FF69B4": "Rosa",
      "#FF0000": "Vermelho",
      "#00FF00": "Verde",
      "#0000FF": "Azul Escuro",
      "#FFFF00": "Amarelo",
      "#FFA500": "Laranja",
      "#800080": "Roxo",
      "#808080": "Cinza",
      "#964B00": "Marrom",
      
      "preto": "Preto",
      "branco": "Branco", 
      "azul": "Azul",
      "rosa": "Rosa",
      "vermelho": "Vermelho",
      "verde": "Verde",
      "amarelo": "Amarelo",
      "laranja": "Laranja",
      "roxo": "Roxo",
      "cinza": "Cinza",
      "marrom": "Marrom",
      
      "black": "Preto",
      "white": "Branco",
      "blue": "Azul",
      "pink": "Rosa", 
      "red": "Vermelho",
      "green": "Verde",
      "yellow": "Amarelo",
      "orange": "Laranja",
      "purple": "Roxo",
      "gray": "Cinza",
      "grey": "Cinza",
      "brown": "Marrom"
    };
    
    const normalizedColor = color.toString().toLowerCase().trim().replace(/\s+/g, '');
    
    if (colorMap[normalizedColor]) {
      return colorMap[normalizedColor];
    }
    
    for (const [key, value] of Object.entries(colorMap)) {
      if (normalizedColor.includes(key.toLowerCase()) || key.toLowerCase().includes(normalizedColor)) {
        return value;
      }
    }
    
    if (/^#([A-Fa-f0-9]{6}|[A-Fa-f0-9]{3})$/.test(color)) {
      return color.toUpperCase();
    }
    
    return color.charAt(0).toUpperCase() + color.slice(1);
  };

  // Function to get all unique sizes and colors from order items
  const getOrderAttributes = (order) => {
    if (!order.items || !Array.isArray(order.items) || order.items.length === 0) {
      return { sizes: [], colors: [], images: [] };
    }

    const sizes = new Set();
    const colors = new Set();
    const images = [];

    order.items.forEach(item => {
      const { size, color, image } = getItemAttributes(item);
      if (size && size !== 'Não especificado') {
        sizes.add(size);
      }
      if (color && color !== 'Não especificada') {
        colors.add(getColorName(color));
      }
      if (image) {
        images.push(image);
      }
    });

    return {
      sizes: Array.from(sizes),
      colors: Array.from(colors),
      images: images.slice(0, 3) // Limit to 3 images for display
    };
  };

  const apiService = {
    async getOrders() {
      try {
        console.log('Buscando pedidos em:', `${API_BASE_URL}/infos`);
        const controller = new AbortController();
        const timeoutId = setTimeout(() => controller.abort(), 10000);

        const response = await fetch(`${API_BASE_URL}/infos`, {
          method: 'GET',
          headers: {
            'Accept': 'application/json',
            'Content-Type': 'application/json',
          },
          signal: controller.signal,
        });

        clearTimeout(timeoutId);

        const contentType = response.headers.get('content-type');
        if (contentType && contentType.includes('text/html')) {
          const text = await response.text();
          throw new Error('Erro no servidor. Verifique os logs do Laravel.');
        }

        if (!response.ok) {
          let errorMessage = `HTTP ${response.status}: ${response.statusText}`;
          try {
            const errorData = await response.json();
            errorMessage = errorData.message || errorData.error || errorMessage;
          } catch {}
          throw new Error(errorMessage);
        }

        const data = await response.json();
        if (!data.success) throw new Error(data.message || 'Erro na API');
        
        return data.data || [];
      } catch (error) {
        if (error.name === 'AbortError') throw new Error('Timeout: servidor demorou');
        if (error.message.includes('Failed to fetch')) throw new Error('Servidor offline ou CORS bloqueado');
        throw error;
      }
    },

    async generateSampleOrders() {
      const response = await fetch(`${API_BASE_URL}/infos/generate-samples`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json', 'Accept': 'application/json' },
        body: JSON.stringify({}),
      });
      if (!response.ok) throw new Error(`Erro ${response.status}`);
      const data = await response.json();
      return data.data;
    },
  };

  const sortOrders = (orders, config) => {
    if (!config.key) return orders;

    return [...orders].sort((a, b) => {
      let aValue = a[config.key];
      let bValue = b[config.key];

      if (aValue == null) aValue = '';
      if (bValue == null) bValue = '';

      const fieldConfig = sortableFields[config.key];
      
      let comparison = 0;
      
      switch (fieldConfig?.type) {
        case 'number':
          aValue = parseFloat(aValue) || 0;
          bValue = parseFloat(bValue) || 0;
          comparison = aValue - bValue;
          break;
        
        case 'date':
          aValue = new Date(aValue || 0);
          bValue = new Date(bValue || 0);
          comparison = aValue - bValue;
          break;
        
        case 'string':
        default:
          aValue = String(aValue).toLowerCase();
          bValue = String(bValue).toLowerCase();
          comparison = aValue.localeCompare(bValue, 'pt-BR');
          break;
      }

      return config.direction === 'desc' ? -comparison : comparison;
    });
  };

  const handleSort = (key) => {
    setSortConfig(currentConfig => ({
      key,
      direction: 
        currentConfig.key === key && currentConfig.direction === 'asc' 
          ? 'desc' 
          : 'asc'
    }));
  };

  const renderSortIcon = (key) => {
    if (sortConfig.key !== key) {
      return <span className="text-gray-400">↕</span>;
    }
    
    return sortConfig.direction === 'asc' 
      ? <FiChevronUp className="text-black" size={16} />
      : <FiChevronDown className="text-black" size={16} />;
  };

  const testApiConnection = async () => {
    setLoading(true);
    try {
      const controller = new AbortController();
      const timeoutId = setTimeout(() => controller.abort(), 5000);
      const response = await fetch(`${API_BASE_URL}/test`, {
        signal: controller.signal,
      });
      clearTimeout(timeoutId);

      const result = response.ok
        ? { connected: true, data: await response.json() }
        : { connected: false, error: `HTTP ${response.status}` };

      setApiStatus(result);
      alert(result.connected ? 'API conectada!' : `Falha: ${result.error}`);
    } catch (err) {
      const msg = err.name === 'AbortError' ? 'Timeout no teste' : err.message;
      setApiStatus({ connected: false, error: msg });
      alert('Erro ao testar conexão');
    } finally {
      setLoading(false);
    }
  };

  const fetchOrders = async () => {
    setLoading(true);
    setError(null);
    try {
      const ordersData = await apiService.getOrders();
      setOrders(ordersData);
    } catch (err) {
      setError("Erro ao carregar pedidos: " + err.message);
      setOrders([]);
    } finally {
      setLoading(false);
    }
  };

  const generateSampleOrders = async () => {
    if (!window.confirm("Deseja gerar pedidos de exemplo para teste?")) return;
    setLoading(true);
    try {
      await apiService.generateSampleOrders();
      await fetchOrders();
    } catch (err) {
      setError("Erro ao gerar exemplo: " + err.message);
      setLoading(false);
    }
  };

  const updateOrderStatus = (orderId, newStatus) => {
    setOrders(prev => prev.map(o => o.id === orderId ? { ...o, status: newStatus } : o));
  };

  const deleteOrder = (orderId) => {
    setOrders(prev => prev.filter(o => o.id !== orderId));
    setDeleteConfirm(null);
  };

  const viewOrderDetails = (order) => {
    console.log('📋 Pedido selecionado para detalhes:', order);
    setSelectedOrder(order);
    setShowOrderModal(true);
  };

  const cancelOrder = (orderId) => {
    if (window.confirm("Tem certeza que deseja cancelar este pedido?")) {
      updateOrderStatus(orderId, "cancelled");
    }
  };

  const confirmDelete = (order) => setDeleteConfirm(order);
  const cancelDelete = () => setDeleteConfirm(null);

  const getStatusText = (status) => {
    const map = {
      pending: "Pendente",
      processing: "Em Preparação",
      shipped: "Enviado",
      delivered: "Entregue",
      cancelled: "Cancelado",
    };
    return map[status] || status;
  };

  const getPaymentText = (payment) => {
    const map = { 
      pix: "PIX", 
      credit_card: "Cartão de Crédito", 
      boleto: "Boleto",
      debit_card: "Cartão de Débito",
      cash: "Dinheiro"
    };
    return map[payment] || payment;
  };

  const getStatusColor = (status) => {
    const colors = {
      shipped: "bg-green-100 text-green-800 border border-green-200",
      processing: "bg-yellow-100 text-yellow-800 border border-yellow-200",
      pending: "bg-blue-100 text-blue-800 border border-blue-200",
      delivered: "bg-purple-100 text-purple-800 border border-purple-200",
      cancelled: "bg-red-100 text-red-800 border border-red-200",
    };
    return colors[status] || "bg-gray-100 text-gray-800 border border-gray-200";
  };

  const getPaymentColor = (payment) => {
    const colors = {
      credit_card: "bg-green-100 text-green-800",
      debit_card: "bg-blue-100 text-blue-800",
      pix: "bg-blue-100 text-blue-800",
      boleto: "bg-purple-100 text-purple-800",
      cash: "bg-gray-100 text-gray-800",
    };
    return colors[payment] || "bg-gray-100 text-gray-800";
  };

  const formatCurrency = (value) => {
    if (value == null || isNaN(value)) return 'R$ 0,00';
    return `R$ ${parseFloat(value).toFixed(2)}`;
  };

  const filteredAndSortedOrders = sortOrders(
    orders.filter(order =>
      order && (
        (order.order_number?.toLowerCase().includes(orderSearch.toLowerCase())) ||
        (order.customer_name?.toLowerCase().includes(orderSearch.toLowerCase())) ||
        (order.customer_email?.toLowerCase().includes(orderSearch.toLowerCase())) ||
        getStatusText(order.status).toLowerCase().includes(orderSearch.toLowerCase()) ||
        getPaymentText(order.payment_method).toLowerCase().includes(orderSearch.toLowerCase()) ||
        formatCurrency(order.total).toLowerCase().includes(orderSearch.toLowerCase())
      )
    ),
    sortConfig
  );

  const getDisplayValue = (order, key) => {
    const value = order[key];
    
    switch (key) {
      case 'total':
        return formatCurrency(value);
      case 'created_at':
        return order.created_at ? new Date(order.created_at).toLocaleDateString('pt-BR') : 'N/A';
      case 'status':
        return getStatusText(value);
      case 'payment_method':
        return getPaymentText(value);
      default:
        return value || 'N/A';
    }
  };

  useEffect(() => {
    fetchOrders();
  }, []);

      <div className="flex flex-col sm:flex-row justify-between items-start sm:items-center gap-4">
        <div>
          <h2 className="text-xl font-semibold">Gerenciamento de Pedidos</h2>
          <div className="text-gray-600 text-sm mt-1">
  {loading ? (
    <span className="inline-flex items-center gap-2">
      <span className="w-4 h-4 bg-gray-200 rounded-full animate-pulse"></span>
      Carregando pedidos...

        <span className="text-blue-600 ml-2">
          • Ordenado por: {sortableFields[sortConfig.key]?.label} ({sortConfig.direction === 'asc' ? 'crescente' : 'decrescente'})
        </span>
      )}
    </>
  )}
</div>
</div>
        <div className="flex items-center gap-3 flex-wrap">

          >

          </button>
          <div className="relative w-full sm:w-64">
            <input
              type="text"

              onChange={(e) => setOrderSearch(e.target.value)}
              disabled={loading}
              className="w-full p-2 pl-10 border border-gray-300 rounded text-black focus:outline-none focus:ring-2 focus:ring-blue-500 disabled:bg-gray-100 disabled:cursor-not-allowed"
            />
            <FiSearch className="absolute left-3 top-2.5 text-gray-400" size={16} />
          </div>
        </div>
      </div>

      <div className="bg-gray-50 p-4 rounded-lg border">
        <div className="flex flex-wrap items-center gap-2">
          <span className="text-sm font-medium text-gray-700">Ordenar por:</span>
          {Object.entries(sortableFields).map(([key, config]) => (

              className={`
                box-border inline-flex items-center gap-1
                px-3 py-1 rounded text-xs font-medium transition-colors border
                focus:outline-none focus:ring-0 disabled:opacity-50 disabled:cursor-not-allowed
                ${
                  sortConfig.key === key
                    ? "bg-blue-600 text-white border-blue-600"
                    : "bg-white text-gray-700 border-gray-300 hover:bg-gray-50"
                }
              `}
            >
              <span className="leading-none">{config.label}</span>
              {renderSortIcon(key)}
            </button>
          ))}

          <button
            onClick={() => setSortConfig({ key: 'created_at', direction: 'desc' })}
            disabled={loading}
            className="box-border inline-flex items-center gap-1 px-3 py-1 rounded text-xs font-medium border bg-gray-200 text-gray-700 border-gray-300 hover:bg-gray-300 focus:outline-none focus:ring-0 disabled:opacity-50 disabled:cursor-not-allowed"
          >
            Resetar

          <table className="min-w-full divide-y divide-gray-200">

                    Pedido
                    {renderSortIcon('order_number')}
                  </div>
                </th>
                <th 
                  className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider cursor-pointer hover:bg-gray-100 transition-colors"
                  onClick={() => handleSort('customer_name')}
                >
                  <div className="flex items-center gap-1">
                    Cliente

                  onClick={() => handleSort('created_at')}
                >
                  <div className="flex items-center gap-1">
                    Data
                    {renderSortIcon('created_at')}
                  </div>

                >
                  <div className="flex items-center gap-1">
                    Total

                  Tamanhos
                </th>
                <th className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">
                  Cores

                  onClick={() => handleSort('payment_method')}
                >
                  <div className="flex items-center gap-1">
                    Pagamento
                    {renderSortIcon('payment_method')}
                  </div>
                </th>
                <th 
                  className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider cursor-pointer hover:bg-gray-100 transition-colors"
                  onClick={() => handleSort('status')}
                >
                  <div className="flex items-center gap-1">
                    Status
                    {renderSortIcon('status')}
                  </div>
                </th>
                <th className="px-4 sm:px-6 py-3 text-left text-xs font-medium text-gray-500 uppercase tracking-wider">
                  Ações
                </th>
              </tr>
            </thead>
            <tbody className="bg-white divide-y divide-gray-200">
              {loading ? (
                // Show skeleton loading rows
                [...Array(6)].map((_, index) => (
                  <SkeletonRow key={index} />
                ))
              ) : filteredAndSortedOrders.length > 0 ? (
                filteredAndSortedOrders.map((order) => {
                  const { sizes, colors, images } = getOrderAttributes(order);
                  
                  return (

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\Log;

class Order extends Model
{
    use HasFactory;

    protected $table = 'infos';

    protected $fillable = [
        'order_number',
        'customer_name',
        'customer_email',
        'customer_phone',
        'customer_cpf',
        'total',
        'subtotal',
        'shipping',
        'discount',
        'payment_method',
        'payment_status',
        'status',
        'shipping_street',
        'shipping_number',
        'shipping_complement',
        'shipping_neighborhood',
        'shipping_city',
        'shipping_state',
        'shipping_zip_code',
        'items',
        'pix_data',
        'metadata'
    ];

    protected $casts = [
        'total' => 'decimal:2',
        'subtotal' => 'decimal:2',
        'shipping' => 'decimal:2',
        'discount' => 'decimal:2',
        'items' => 'array',
        'pix_data' => 'array',
        'metadata' => 'array',
        'created_at' => 'datetime',
        'updated_at' => 'datetime'
    ];

    /**
     * Boot method to handle JSON encoding issues
     */
    protected static function boot()
    {
        parent::boot();

        static::retrieved(function ($order) {
            // Ensure JSON fields are properly handled
            try {
                if ($order->items && is_string($order->items)) {
                    $order->items = json_decode($order->items, true);
                }
                if ($order->pix_data && is_string($order->pix_data)) {
                    $order->pix_data = json_decode($order->pix_data, true);
                }
                if ($order->metadata && is_string($order->metadata)) {
                    $order->metadata = json_decode($order->metadata, true);
                }
            } catch (\Exception $e) {
                Log::error('Error decoding JSON fields for order ' . $order->id . ': ' . $e->getMessage());
            }
        });
    }
}

    /**
     * Update info status and payment status
     * PUT /api/infos/{id}/status
     */
    public function updateStatus($id, Request $request): JsonResponse
    {
        try {
            Log::info('=== INFO STATUS UPDATE START ===', ['info_id' => $id]);

            // Validate input
            $validated = $request->validate([
                'status' => 'required|string|in:pending,processing,shipped,delivered,cancelled',
                'payment_status' => 'required|string|in:pending,paid,failed,refunded'
            ]);

            // Find info
            $info = Order::find($id);
            if (!$info) {
                return response()->json([
                    'success' => false,
                    'message' => 'Info not found',
                    'debug' => ['info_id' => $id]
                ], 404);
            }

            // Update fields
            $info->status = $validated['status'];
            $info->payment_status = $validated['payment_status'];
            $info->save();

            Log::info('Info status updated successfully', [
                'info_id' => $info->id,
                'status' => $info->status,
                'payment_status' => $info->payment_status
            ]);

            return response()->json([
                'success' => true,
                'message' => 'Info status updated successfully',
                'data' => [
                    'info_id' => $info->id,
                    'order_number' => $info->order_number,
                    'status' => $info->status,
                    'payment_status' => $info->payment_status
                ],
                'debug' => [
                    'timestamp' => now()->format('c')
                ]
            ]);

        } catch (Exception $e) {
            Log::error('=== INFO STATUS UPDATE ERROR ===');
            Log::error('Error: ' . $e->getMessage());
            Log::error('File: ' . $e->getFile() . ':' . $e->getLine());
            Log::error('Request data: ', $request->all());

            return response()->json([
                'success' => false,
                'message' => 'Failed to update info status',
                'error' => $e->getMessage(),
                'debug' => [
                    'exception' => get_class($e),
                    'message' => $e->getMessage(),

const CsrfTokenManager = {
  getCsrfToken: async () => {
    try {
      const response = await fetch('http://localhost:8000/sanctum/csrf-cookie', {
        method: 'GET',
        credentials: 'include',
      });

      if (response.ok) {
        const cookies = document.cookie;
        const tokenMatch = cookies.match(/XSRF-TOKEN=([^;]+)/);
        
        if (tokenMatch && tokenMatch[1]) {
          return decodeURIComponent(tokenMatch[1]);
        }
      }
      
      const metaToken = document.querySelector('meta[name="csrf-token"]')?.getAttribute('content');
      return metaToken || 'fallback-csrf-token';
      
    } catch (error) {
      console.error('❌ Error getting CSRF token:', error);
      return 'fallback-csrf-token';
    }
  }

<?php

class CartController extends Controller

    /**
     * Get user's cart with product details
     */
    public function index()
    {
        try {
            $user = Auth::user();
            if (!$user) {
                Log::warning('Unauthorized cart access attempt');
                return response()->json(['error' => 'Unauthorized'], 401);
            }

            Log::info('Fetching cart for user:', ['user_id' => $user->id]);

            // Check MongoDB connection
            if (!$this->mongo->isConnected()) {
                Log::error('MongoDB not connected');
                return response()->json(['error' => 'Database connection failed'], 500);
            }

            $cartCollection = $this->mongo->getCartCollection();
            $productsCollection = $this->mongo->getProductsCollection();

            // Find cart items for user
            $cartItems = $cartCollection->find(['user_id' => (string)$user->id]);
            
            $cartWithProducts = [];
            $total = 0;
            
            foreach ($cartItems as $item) {
                try {
                    // Convert BSON document to array
                    $itemArray = json_decode(json_encode($item), true);
                    
                    // Handle product_id - it might be stored as ObjectId or string
                    $productId = $itemArray['product_id'];
                    
                    // If product_id is an array (BSON ObjectId), convert to string
                    if (is_array($productId) && isset($productId['$oid'])) {
                        $productId = $productId['$oid'];
                    }
                    
                    Log::info('Processing cart item:', [
                        'item_id' => $itemArray['_id']['$oid'] ?? $itemArray['_id'],
                        'product_id' => $productId,
                        'quantity' => $itemArray['quantity'],
                        'size' => $itemArray['size'] ?? 'N/A',
                        'color' => $itemArray['color'] ?? 'N/A'
                    ]);

                    // Find product - handle both ObjectId and string formats
                    $product = null;
                    if (!empty($productId)) {
                        try {
                            // Try to find product by ObjectId
                            $product = $productsCollection->findOne(['_id' => new ObjectId($productId)]);
                            
                            if (!$product) {
                                // If not found, try as string
                                $product = $productsCollection->findOne(['_id' => $productId]);
                            }
                        } catch (\Exception $e) {
                            Log::warning('Error finding product:', [
                                'product_id' => $productId,
                                'error' => $e->getMessage()
                            ]);
                        }
                    }

                    // Convert product to array if found
                    $productArray = null;
                    if ($product) {
                        $productArray = json_decode(json_encode($product), true);
                        
                        // Ensure _id is converted to string
                        if (isset($productArray['_id']) && is_array($productArray['_id']) && isset($productArray['_id']['$oid'])) {
                            $productArray['id'] = $productArray['_id']['$oid'];
                        } else {
                            $productArray['id'] = (string)$productArray['_id'];
                        }
                        
                        // Ensure consistent field names
                        $productArray['name'] = $productArray['name'] ?? $productArray['title'] ?? 'Produto sem nome';
                        $productArray['price'] = floatval($productArray['price'] ?? 0);
                        $productArray['image'] = $productArray['image'] ?? $productArray['image_url'] ?? null;
                        $productArray['description'] = $productArray['description'] ?? 'Descrição não disponível';
                        $productArray['category'] = $productArray['category'] ?? 'Geral';
                        
                        // Calculate item total
                        $itemTotal = $productArray['price'] * $itemArray['quantity'];
                        $total += $itemTotal;
                    }

                    // Build the cart item with proper structure including size and color
                    $cartItem = [
                        'id' => $itemArray['_id']['$oid'] ?? (string)$itemArray['_id'],
                        'product_id' => $productId,
                        'user_id' => $itemArray['user_id'],
                        'quantity' => intval($itemArray['quantity']),
                        'size' => $itemArray['size'] ?? null,
                        'color' => $itemArray['color'] ?? null,
                        'created_at' => $itemArray['created_at'] ?? null,
                        'updated_at' => $itemArray['updated_at'] ?? null,
                        'product' => $productArray
                    ];

                    $cartWithProducts[] = $cartItem;
                    
                    Log::debug('Processed cart item:', $cartItem);

                } catch (\Exception $itemError) {
                    Log::error('Error processing cart item:', [
                        'item' => $item,
                        'error' => $itemError->getMessage()
                    ]);
                    continue; // Skip this item but continue with others
                }
            }

            Log::info('Successfully processed cart items:', [
                'count' => count($cartWithProducts),
                'total' => $total
            ]);

            return response()->json([
                'cart' => $cartWithProducts,
                'count' => count($cartWithProducts),
                'total' => $total,
                'success' => true
            ]);

        } catch (\Exception $e) {
            Log::error('Cart index error:', [
                'error' => $e->getMessage(),
                'file' => $e->getFile(),
                'line' => $e->getLine(),
                'trace' => $e->getTraceAsString()
            ]);
            return response()->json([
                'error' => 'Failed to fetch cart: ' . $e->getMessage(),
                'success' => false
            ], 500);
        }
    }

    /**
     * Add product to cart or update quantity - UPDATED FOR SIZE/COLOR SUPPORT
     */
    public function addToCart(Request $request)
    {
        try {
            $user = Auth::user();
            if (!$user) {
                Log::warning('Unauthorized cart add attempt');
                return response()->json(['error' => 'Unauthorized'], 401);
            }

            $request->validate([
                'product_id' => 'required|string',
                'quantity' => 'required|integer|min:1',
                'size' => 'required|string',
                'color' => 'required|string'
            ]);

            $productId = $request->product_id;
            $quantity = $request->quantity;
            $size = $request->size;
            $color = $request->color;

            Log::info('Adding product to cart with attributes:', [
                'user_id' => $user->id,
                'product_id' => $productId,
                'quantity' => $quantity,
                'size' => $size,
                'color' => $color
            ]);

            // Check MongoDB connection
            if (!$this->mongo->isConnected()) {
                Log::error('MongoDB not connected');
                return response()->json(['error' => 'Database connection failed'], 500);
            }

            $cartCollection = $this->mongo->getCartCollection();
            $productsCollection = $this->mongo->getProductsCollection();

            // Check if product exists
            $product = $productsCollection->findOne(['_id' => new ObjectId($productId)]);
            if (!$product) {
                return response()->json(['error' => 'Product not found'], 404);
            }

            // ✅ UPDATED: Check if EXACT same item exists (same product + same size + same color)
            $existingItem = $cartCollection->findOne([
                'user_id' => (string)$user->id,
                'product_id' => $productId,
                'size' => $size,
                'color' => $color
            ]);

            $now = new \MongoDB\BSON\UTCDateTime(now()->getTimestamp() * 1000);

            if ($existingItem) {
                // ✅ Update quantity only for EXACT same product with same attributes
                $newQuantity = $existingItem['quantity'] + $quantity;
                $cartCollection->updateOne(
                    ['_id' => $existingItem['_id']],
                    [
                        '$set' => [
                            'quantity' => $newQuantity,
                            'updated_at' => $now
                        ]
                    ]
                );

                Log::info('Cart item quantity updated (same attributes):', [
                    'user_id' => $user->id,
                    'product_id' => $productId,
                    'size' => $size,
                    'color' => $color,
                    'old_quantity' => $existingItem['quantity'],
                    'new_quantity' => $newQuantity
                ]);

                return response()->json([
                    'message' => 'Cart item quantity updated',
                    'product_id' => $productId,
                    'quantity' => $newQuantity,
                    'size' => $size,
                    'color' => $color,
                    'action' => 'updated'
                ]);
            } else {
                // ✅ Add NEW item to cart (even if same product but different size/color)
                $insertResult = $cartCollection->insertOne([
                    'user_id' => (string)$user->id,
                    'product_id' => $productId,
                    'quantity' => $quantity,
                    'size' => $size,
                    'color' => $color,
                    'created_at' => $now,
                    'updated_at' => $now
                ]);

                Log::info('New product variant added to cart:', [
                    'user_id' => $user->id,
                    'product_id' => $productId,
                    'quantity' => $quantity,
                    'size' => $size,
                    'color' => $color,
                    'cart_item_id' => (string)$insertResult->getInsertedId()
                ]);

                return response()->json([
                    'message' => 'Product added to cart',
                    'product_id' => $productId,
                    'quantity' => $quantity,
                    'size' => $size,
                    'color' => $color,
                    'action' => 'added'
                ]);
            }

        } catch (\Illuminate\Validation\ValidationException $e) {
            Log::error('Validation error in addToCart', ['errors' => $e->errors()]);
            return response()->json(['error' => 'Validation failed', 'messages' => $e->errors()], 422);
        } catch (\Exception $e) {
            Log::error('Add to cart error:', [
                'error' => $e->getMessage(),
                'file' => $e->getFile(),
                'line' => $e->getLine()
            ]);
            return response()->json(['error' => 'Failed to add product to cart: ' . $e->getMessage()], 500);
        }
    }

    /**
     * Update cart item quantity - UPDATED FOR SIZE/COLOR SUPPORT
     */
    public function updateQuantity(Request $request, $itemId)
    {
        try {
            $user = Auth::user();
            if (!$user) {
                return response()->json(['error' => 'Unauthorized'], 401);
            }

            $request->validate([
                'quantity' => 'required|integer|min:0' // Allow 0 to remove item
            ]);

            $quantity = $request->quantity;

            // Check MongoDB connection
            if (!$this->mongo->isConnected()) {
                return response()->json(['error' => 'Database connection failed'], 500);
            }

            $cartCollection = $this->mongo->getCartCollection();

            // ✅ UPDATED: Use item ID instead of product_id to target specific cart item
            if ($quantity === 0) {
                // Remove item if quantity is 0
                $result = $cartCollection->deleteOne([
                    '_id' => new ObjectId($itemId),
                    'user_id' => (string)$user->id
                ]);

                if ($result->getDeletedCount() === 0) {
                    return response()->json(['error' => 'Cart item not found'], 404);
                }

                Log::info('Cart item removed (quantity 0):', [
                    'user_id' => $user->id,
                    'item_id' => $itemId
                ]);

                return response()->json([
                    'message' => 'Item removed from cart',
                    'item_id' => $itemId,
                    'action' => 'removed'
                ]);
            } else {
                // Update quantity
                $result = $cartCollection->updateOne(
                    [
                        '_id' => new ObjectId($itemId),
                        'user_id' => (string)$user->id
                    ],
                    [
                        '$set' => [
                            'quantity' => $quantity,
                            'updated_at' => new \MongoDB\BSON\UTCDateTime(now()->getTimestamp() * 1000)
                        ]
                    ]
                );

                if ($result->getModifiedCount() === 0) {
                    return response()->json(['error' => 'Cart item not found'], 404);
                }

                Log::info('Cart quantity updated:', [
                    'user_id' => $user->id,
                    'item_id' => $itemId,
                    'quantity' => $quantity
                ]);

                return response()->json([
                    'message' => 'Cart quantity updated successfully',
                    'item_id' => $itemId,
                    'quantity' => $quantity,
                    'action' => 'updated'
                ]);
            }

        } catch (\Illuminate\Validation\ValidationException $e) {
            return response()->json(['error' => 'Validation failed', 'messages' => $e->errors()], 422);
        } catch (\Exception $e) {
            Log::error('Update cart quantity error:', ['error' => $e->getMessage()]);
            return response()->json(['error' => 'Failed to update cart quantity'], 500);
        }
    }

    /**

            if (!$this->mongo->isConnected()) {

            return response()->json([
                'message' => 'Cart cleared successfully',
                'deleted_count' => $result->getDeletedCount()
            ]);

        } catch (\Exception $e) {
            Log::error('Clear cart error:', ['error' => $e->getMessage()]);
            return response()->json(['error' => 'Failed to clear cart'], 500);
        }
    }

    /**
     * Get cart count
     */
    public function count()
    {
        try {
            $user = Auth::user();
            if (!$user) return response()->json(['error' => 'Unauthorized'], 401);

            $cartCollection = $this->mongo->getCartCollection();
            $count = $cartCollection->countDocuments(['user_id' => (string)$user->id]);

            return response()->json(['count' => $count]);
        } catch (\Exception $e) {
            Log::error('Cart count error:', ['error' => $e->getMessage()]);
            return response()->json(['error' => 'Failed to get cart count'], 500);
        }
    }

    /**
     * Debug endpoint to check raw cart data
     */
    public function debug()
    {
        try {
            $user = Auth::user();
            if (!$user) return response()->json(['error' => 'Unauthorized'], 401);

            $cartCollection = $this->mongo->getCartCollection();
            $productsCollection = $this->mongo->getProductsCollection();

            // Get raw cart items
            $cartItems = $cartCollection->find(['user_id' => (string)$user->id])->toArray();
            $rawItems = [];
            
            foreach ($cartItems as $item) {
                $rawItems[] = json_decode(json_encode($item), true);
            }

            return response()->json([
                'cart_raw' => $rawItems,
                'user_id' => $user->id,
                'message' => 'Size and color attributes are now supported for duplicate products'
            ]);

        } catch (\Exception $e) {
            return response()->json(['error' => $e->getMessage()], 500);
        }
    }

    /**
     * NEW: Get cart items by product ID (for checking variants)
     */
    public function getProductVariants($productId)
    {
        try {
            $user = Auth::user();
            if (!$user) return response()->json(['error' => 'Unauthorized'], 401);

            $cartCollection = $this->mongo->getCartCollection();
            
            $variants = $cartCollection->find([
                'user_id' => (string)$user->id,
                'product_id' => $productId
            ])->toArray();

            $variantData = [];
            foreach ($variants as $variant) {
                $variantData[] = [
                    'item_id' => (string)$variant['_id'],
                    'size' => $variant['size'] ?? null,
                    'color' => $variant['color'] ?? null,
                    'quantity' => $variant['quantity']
                ];
            }

            return response()->json([
                'product_id' => $productId,
                'variants_count' => count($variantData),
                'variants' => $variantData
            ]);

        } catch (\Exception $e) {
            return response()->json(['error' => $e->getMessage()], 500);
        }
    }

const CsrfTokenManager = {
  getCsrfToken: async () => {
    try {
      console.log('🔄 Fetching CSRF token...');
      
      const response = await fetch('http://localhost:8000/sanctum/csrf-cookie', {
        method: 'GET',
        credentials: 'include',
      });

      if (response.ok) {
        const cookies = document.cookie;
        const tokenMatch = cookies.match(/XSRF-TOKEN=([^;]+)/);
        
        if (tokenMatch && tokenMatch[1]) {
          const token = decodeURIComponent(tokenMatch[1]);
          console.log('✅ CSRF token obtained');
          return token;
        }
      }
      
      const metaToken = document.querySelector('meta[name="csrf-token"]')?.getAttribute('content');
      return metaToken || 'fallback-csrf-token';
      
    } catch (error) {
      console.error('❌ Error getting CSRF token:', error);
      return 'fallback-csrf-token';
    }
  }
};

// ✅ CEP Input Component
const CepInput = ({ 
  cep, 
  onCepChange, 
  loading, 
  shippingData,
  onAutoFillAddress 
}) => {
  const [isValidCep, setIsValidCep] = useState(false);
  const [showError, setShowError] = useState(false);

  const handleCepChange = (e) => {
    const value = e.target.value.replace(/\D/g, '');
    let formattedCep = value;
    
    if (value.length > 5) {
      formattedCep = value.substring(0, 5) + '-' + value.substring(5, 8);
    }
    
    onCepChange(formattedCep);
    
    const hasValidLength = value.length === 8;
    setIsValidCep(hasValidLength);
    
    if (value.length > 0 && !hasValidLength && value.length < 8) {
      setShowError(false);
    } else {
      setShowError(value.length > 0 && !hasValidLength);
    }
  };

  const handleCepBlur = () => {
    const cleanCep = cep.replace(/\D/g, '');
    const hasValidLength = cleanCep.length === 8;
    
    setShowError(cleanCep.length > 0 && !hasValidLength);
    
    if (hasValidLength) {
      onAutoFillAddress(cep);
    }
  };

  const getMissingDigits = () => {
    const cleanCep = cep.replace(/\D/g, '');
    return 8 - cleanCep.length;
  };

  return (
    <div className="bg-white p-4 rounded-lg border border-gray-200 mb-6">
      <div className="flex items-center gap-2 mb-3">
        <FiMapPin className="text-blue-600" size={20} />
        <h3 className="text-lg font-semibold text-gray-800">Calcular Frete</h3>
      </div>
      
      <div className="flex gap-3">
        <div className="flex-1">
          <label className="block text-sm font-medium text-gray-700 mb-2">
            Digite seu CEP
          </label>
          <div className="relative">
            <input
              type="text"
              value={cep}
              onChange={handleCepChange}
              onBlur={handleCepBlur}
              className="w-full p-3 border border-gray-300 rounded-lg focus:ring-2 focus:ring-blue-500 focus:border-transparent text-black pr-24"
              placeholder="00000-000"
              maxLength={9}
            />
            <div className="absolute right-2 top-1/2 transform -translate-y-1/2">
              {loading && (
                <div className="animate-spin rounded-full h-5 w-5 border-b-2 border-blue-600"></div>
              )}
            </div>
          </div>
          
          {showError && (
            <p className="text-red-500 text-xs mt-1">CEP deve ter 8 dígitos</p>
          )}
          
          {cep && cep.replace(/\D/g, '').length > 0 && cep.replace(/\D/g, '').length < 8 && !showError && (
            <p className="text-gray-500 text-xs mt-1">
              Digite mais {getMissingDigits()} dígito{getMissingDigits() !== 1 ? 's' : ''}
            </p>
          )}
        </div>
      </div>

      {shippingData.cep && isValidCep && (
        <div className="mt-3 p-3 bg-green-50 border border-green-200 rounded-lg">
          <div className="flex items-center gap-2">
            <FiCheck className="text-green-600" size={16} />
            <span className="text-green-800 text-sm font-medium">
              CEP {shippingData.cep} válido
            </span>
          </div>
          {shippingData.city && shippingData.state && (
            <p className="text-green-700 text-sm mt-1">
              {shippingData.city} - {shippingData.state}
            </p>
          )}
        </div>
      )}
    </div>
  );

          console.error('Error updating database:', error);

      setTimeLeft(prev => {
        if (prev <= 1) {
          clearInterval(timer);
          return 0;

        return prev - 1;

      clearInterval(timer);
      clearTimeout(autoSuccessTimer);

                      </div>

                <div className="flex items-center justify-between">
                  <div>

            Dados Pessoais

            Entrega
          </button>

            Pagamento
          </button>

            >

        </div>
      </div>
    </div>

                userAddresses={userAddresses}
                selectedAddressId={selectedAddressId}
                handleAddressSelect={handleAddressSelect}
                fetchUserAddresses={fetchUserAddresses}
                selectedShipping={selectedShipping}
                onShippingSelect={handleShippingSelect}
                cartItems={cartItems}
                prevStep={prevStep}
                nextStep={nextStep}
              />
            )}
            {currentStep === 3 && (
              <PaymentStep 
                paymentMethod={paymentMethod}
                setPaymentMethod={setPaymentMethod}
                cardData={cardData}
                setCardData={setCardData}
                calculateTotal={calculateTotal}
                prevStep={prevStep}
                processCheckout={processCheckout}
                loading={loading}
              />
            )}
            {currentStep === 4 && (
              <PixStep 
                pixData={pixData}
                copyPixCode={copyPixCode}
                goToStep={goToStep}
              />
            )}
          </div>

          <div className="lg:col-span-1">
            <OrderSummary 
              cartItems={cartItems}
              couponCode={couponCode}
              setCouponCode={setCouponCode}
              applyCoupon={applyCoupon}
              calculateSubtotal={calculateSubtotal}
              calculateShipping={calculateShipping}
              discount={discount}
              calculateTotal={calculateTotal}
              paymentMethod={paymentMethod}
              currentStep={currentStep}
              goToStep={goToStep}
              selectedShipping={selectedShipping}
              isUsingSelectedItems={isUsingSelectedItems()}
              navigate={navigate}
            />
          </div>
        </div>
      </div>
    </div>
  );
};

export default CheckoutPage;
