Wishlist.php

<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Wishlist extends Model
{
    use HasFactory;

    protected $table = 'wishlist';

    protected $fillable = [
        'user_id',
        '_id'
    ];

    protected $casts = [
        '_id' => 'string',
    ];

    /**
     * Get the user that owns the wishlist item
     */
    public function user()
    {
        return $this->belongsTo(User::class);
    }

    /**
     * Get the product that is in wishlist
     */
    public function product()
    {
        return $this->belongsTo(Product::class, '_id');
    }

    /**
     * Check if product is already in wishlist
     */
    public static function isProductInWishlist($userId, $productId)
    {
        return static::where('user_id', $userId)
                    ->where('_id', $productId)
                    ->exists();
    }
}

WishlistController.php

<?php

namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Auth;
use Illuminate\Support\Facades\Log;
use App\Services\MongoDatabaseService;

class WishlistController extends Controller
{
    protected $mongo;

    public function __construct(MongoDatabaseService $mongo)
    {
        $this->mongo = $mongo;
    }

    /**
     * Get user's wishlist with product details
     */
    public function index()
    {
        try {
            $user = Auth::user();
            if (!$user) return response()->json(['error' => 'Unauthorized'], 401);

            $wishlistCollection = $this->mongo->getWishlistCollection();
            $productsCollection = $this->mongo->getProductsCollection();

            $wishlistItems = $wishlistCollection->find(['user_id' => (string)$user->id])->toArray();

            // Fetch product details
            $wishlistWithProducts = array_map(function ($item) use ($productsCollection) {
                $product = $productsCollection->findOne(['_id' => $item['product_id']]);
                $item['product'] = $product ?? null;
                return $item;
            }, $wishlistItems);

            return response()->json([
                'wishlist' => $wishlistWithProducts,
                'count' => count($wishlistWithProducts),
            ]);
        } catch (\Exception $e) {
            Log::error('Wishlist index error:', ['error' => $e->getMessage()]);
            return response()->json(['error' => 'Failed to fetch wishlist'], 500);
        }
    }

    /**
     * Toggle product in wishlist (add/remove)
     */
    public function toggle(Request $request)
    {
        try {
            $user = Auth::user();
            if (!$user) return response()->json(['error' => 'Unauthorized'], 401);

            $request->validate(['product_id' => 'required|string']);
            $productId = $request->product_id;

            $wishlist = $this->mongo->getWishlistCollection();

            $existing = $wishlist->findOne([
                'user_id' => (string)$user->id,
                'product_id' => $productId,
            ]);

            if ($existing) {
                $wishlist->deleteOne(['_id' => $existing['_id']]);

                return response()->json([
                    'message' => 'Product removed from wishlist',
                    'action' => 'removed',
                    'product_id' => $productId,
                    'in_wishlist' => false,
                ]);
            } else {
                $wishlist->insertOne([
                    'user_id' => (string)$user->id,
                    'product_id' => $productId,
                    'created_at' => new \MongoDB\BSON\UTCDateTime()
                ]);

                return response()->json([
                    'message' => 'Product added to wishlist',
                    'action' => 'added',
                    'product_id' => $productId,
                    'in_wishlist' => true,
                ]);
            }
        } catch (\Illuminate\Validation\ValidationException $e) {
            return response()->json(['error' => 'Validation failed', 'messages' => $e->errors()], 422);
        } catch (\Exception $e) {
            Log::error('Wishlist toggle error:', ['error' => $e->getMessage()]);
            return response()->json(['error' => 'Failed to toggle wishlist'], 500);
        }
    }

    /**
     * Check if product is in wishlist
     */
    public function check($productId)
    {
        try {
            $user = Auth::user();
            if (!$user) return response()->json(['error' => 'Unauthorized'], 401);

            $wishlist = $this->mongo->getWishlistCollection();
            $existing = $wishlist->findOne([
                'user_id' => (string)$user->id,
                'product_id' => $productId,
            ]);

            return response()->json([
                'product_id' => $productId,
                'in_wishlist' => $existing !== null,
            ]);
        } catch (\Exception $e) {
            Log::error('Wishlist check error:', ['error' => $e->getMessage()]);
            return response()->json(['error' => 'Failed to check wishlist status'], 500);
        }
    }

    /**
     * Remove product from wishlist
     */
    public function destroy($productId)
    {
        try {
            $user = Auth::user();
            if (!$user) return response()->json(['error' => 'Unauthorized'], 401);

            $wishlist = $this->mongo->getWishlistCollection();
            $wishlist->deleteOne([
                'user_id' => (string)$user->id,
                'product_id' => $productId,
            ]);

            return response()->json([
                'message' => 'Product removed from wishlist',
                'product_id' => $productId,
            ]);
        } catch (\Exception $e) {
            Log::error('Wishlist destroy error:', ['error' => $e->getMessage()]);
            return response()->json(['error' => 'Failed to remove product from wishlist'], 500);
        }
    }

    /**
     * Get wishlist count
     */
    public function count()
    {
        try {
            $user = Auth::user();
            if (!$user) return response()->json(['error' => 'Unauthorized'], 401);

            $wishlist = $this->mongo->getWishlistCollection();
            $count = $wishlist->countDocuments(['user_id' => (string)$user->id]);

            return response()->json(['count' => $count]);
        } catch (\Exception $e) {
            Log::error('Wishlist count error:', ['error' => $e->getMessage()]);
            return response()->json(['error' => 'Failed to get wishlist count'], 500);
        }
    }
}

MongoDatabaseService.php

<?php

namespace App\Services;

use MongoDB\Client;
use Exception;
use Illuminate\Support\Facades\Log;

class MongoDatabaseService
{
    private $client;
    private $database;
    private $connected = false;

    public function __construct()
    {
        try {
            $connectionString = env('MONGO_DB_URI', 'mongodb://localhost:27017');
            $databaseName = env('MONGO_DB_DATABASE', 'finito');
            
            $this->client = new Client($connectionString);
            $this->database = $this->client->selectDatabase($databaseName);
            $this->connected = true;
            
            Log::info('MongoDB Service initialized successfully');
        } catch (Exception $e) {
            Log::error('MongoDB connection failed: ' . $e->getMessage());
            $this->connected = false;
        }
    }

    public function isConnected(): bool
    {
        return $this->connected;
    }

    public function getCollection(string $collectionName)
    {
        if (!$this->connected) {
            throw new Exception('MongoDB not connected');
        }
        return $this->database->selectCollection($collectionName);
    }

    public function getWishlistCollection()
    {
        return $this->getCollection('wishlist');
    }

    public function getProductsCollection()
    {
        return $this->getCollection('products');
    }
}
