<?php

class MongoWishlist
{
    protected $collection;
    protected $mongoProduct;

    public function __construct()
    {
        $client = new Client(env('MONGODB_URI', 'mongodb://localhost:27017'));
        $database = env('MONGODB_DATABASE', 'finito');
        $this->collection = $client->$database->wishlists;
        $this->mongoProduct = new MongoProduct();
    }

    /**
     * Get user's wishlist with product details
     */
    public function getUserWishlist($userId)
    {
        try {
            // Get wishlist items for user
            $wishlistItems = $this->collection->find([
                'user_id' => (int)$userId
            ])->toArray();

            // Get product IDs from wishlist
            $productIds = array_map(function($item) {
                return $item['product_id'];
            }, $wishlistItems);

            // Get products from MongoDB using MongoProduct
            $products = $this->mongoProduct->getByIds($productIds);

            // Combine wishlist items with product data
            $result = [];
            foreach ($wishlistItems as $item) {
                $product = $products[$item['product_id']] ?? null;
                
                $result[] = [
                    'id' => (string)$item['_id'],
                    'product_id' => $item['product_id'],
                    'user_id' => $item['user_id'],
                    'created_at' => $item['created_at'] ?? null,
                    'updated_at' => $item['updated_at'] ?? null,
                    'product' => $product
                ];
            }

            return $result;

        } catch (\Exception $e) {
            Log::error('MongoWishlist getUserWishlist error:', ['error' => $e->getMessage()]);
            return [];
        }
    }

    /**
     * Check if product is in user's wishlist
     */
    public function isProductInWishlist($userId, $productId)
    {
        try {
            $count = $this->collection->countDocuments([
                'user_id' => (int)$userId,
                'product_id' => $productId
            ]);

            return $count > 0;

        } catch (\Exception $e) {
            Log::error('MongoWishlist isProductInWishlist error:', ['error' => $e->getMessage()]);
            return false;
        }
    }

    /**
     * Add product to wishlist
     */
    public function addToWishlist($userId, $productId)
    {
        try {
            // Check if product exists
            $product = $this->getProduct($productId);
            if (!$product) {
                return ['error' => 'Product not found'];
            }

            // Check if already in wishlist
            if ($this->isProductInWishlist($userId, $productId)) {
                return ['error' => 'Product already in wishlist'];
            }

            $result = $this->collection->insertOne([
                'user_id' => (int)$userId,
                'product_id' => $productId,
                'created_at' => now()->toISOString(),
                'updated_at' => now()->toISOString()
            ]);

            return [
                'success' => true,
                'inserted_id' => (string)$result->getInsertedId()
            ];

        } catch (\Exception $e) {
            Log::error('MongoWishlist addToWishlist error:', ['error' => $e->getMessage()]);
            return ['error' => 'Failed to add to wishlist'];
        }
    }

    /**
     * Remove product from wishlist
     */
    public function removeFromWishlist($userId, $productId)
    {
        try {
            $result = $this->collection->deleteOne([
                'user_id' => (int)$userId,
                'product_id' => $productId
            ]);

            if ($result->getDeletedCount() === 0) {
                return ['error' => 'Product not found in wishlist'];
            }

            return ['success' => true, 'deleted_count' => $result->getDeletedCount()];

        } catch (\Exception $e) {
            Log::error('MongoWishlist removeFromWishlist error:', ['error' => $e->getMessage()]);
            return ['error' => 'Failed to remove from wishlist'];
        }
    }

    /**
     * Toggle product in wishlist
     */
    public function toggleWishlist($userId, $productId)
    {
        try {
            // Check if product exists
            $product = $this->getProduct($productId);
            if (!$product) {
                return ['error' => 'Product not found'];
            }

            if ($this->isProductInWishlist($userId, $productId)) {
                // Remove from wishlist
                $result = $this->removeFromWishlist($userId, $productId);
                if (isset($result['success'])) {
                    return [
                        'action' => 'removed',
                        'product_id' => $productId,
                        'in_wishlist' => false
                    ];
                }
                return $result;
            } else {
                // Add to wishlist
                $result = $this->addToWishlist($userId, $productId);
                if (isset($result['success'])) {
                    return [
                        'action' => 'added',
                        'product_id' => $productId,
                        'in_wishlist' => true
                    ];
                }
                return $result;
            }

        } catch (\Exception $e) {
            Log::error('MongoWishlist toggleWishlist error:', ['error' => $e->getMessage()]);
            return ['error' => 'Failed to toggle wishlist'];
        }
    }

    {
        try {
            $user = Auth::user();
            
            if (!$user) {
                return response()->json(['error' => 'Unauthorized'], 401);

class MongoProductController extends Controller

    /**
     * List all products
     */
    public function index(Request $request)
    {
        try {
            $orderBy = $request->get('orderBy', 'created_at');
            $direction = $request->get('direction', 'desc');
            
            $products = $this->mongoProduct->getAll($orderBy, $direction);

            return response()->json($products);
        } catch (\Exception $e) {
            return response()->json([
                'error' => 'Failed to fetch products',
                'message' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Store a new product with file upload
     */
    public function store(Request $request)
    {
        $validated = $request->validate([
            'title' => 'required|string|max:255',
            'price' => 'required|numeric|min:0',
            'image' => 'required|image|mimes:jpeg,png,jpg,gif,webp|max:5120',
            'category' => 'nullable|string|max:100',
            'description' => 'nullable|string|max:1000',
        ]);

        try {
            $product = $this->mongoProduct->create($validated);

            if (!$product) {
                return response()->json([
                    'error' => 'Failed to create product'
                ], 500);
            }

            return response()->json($product, 201);
        } catch (\Exception $e) {
            return response()->json([
                'error' => 'Failed to create product',
                'message' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Show a single product by ID
     */
    public function show($id)
    {
        try {
            $product = $this->mongoProduct->find($id);

            if (!$product) {
                return response()->json([
                    'error' => 'Product not found'
                ], 404);
            }

            return response()->json($product);
        } catch (\Exception $e) {
            return response()->json([
                'error' => 'Failed to fetch product',
                'message' => $e->getMessage()
            ], 500);
        }
    }

    /**
     * Update a product
     */
    public function update(Request $request, $id)
    {
        $validated = $request->validate([
            'title' => 'sometimes|required|string|max:255',
            'price' => 'sometimes|required|numeric|min:0',
            'image' => 'sometimes|image|mimes:jpeg,png,jpg,gif,webp|max:5120',
            'category' => 'nullable|string|max:100',
            'description' => 'nullable|string|max:1000',
        ]);

        try {
            $product = $this->mongoProduct->update($id, $validated);

            if (!$product) {
                return response()->json([
                    'error' => 'Product not found or no changes made'
                ], 404);
            }

            return response()->json($product);
        } catch (\Exception $e) {
            return response()->json([
                'error' => 'Failed to update product',
                'message' => $e->getMessage()
            ], 500);
        }

              name: 'Product ' + item.product_id,
              price: 0,
              image: '',
              description: 'Product information not available',
              category: 'Unknown'
            };

            const itemTotal = product.price * item.quantity;

            return (
              <Card key={item.id} className="p-4 hover:shadow-md transition-shadow">
                <div className="flex flex-col sm:flex-row gap-4">
                  {/* Product Image */}
                  <div className="flex-shrink-0 w-full sm:w-20 h-40 sm:h-20">
                    {product.image ? (
                      <img 
                        src={product.image} 
                        alt={product.name}
                        className="w-full h-full object-cover rounded-lg"
                      />
                    ) : (
                      <div className="w-full h-full bg-gray-200 rounded-lg flex items-center justify-center">
                        <span className="text-gray-500 text-xs">No image</span>
                      </div>
                    )}
                  </div>

                  {/* Product Info */}
                  <div className="flex-1 min-w-0">
                    <div className="flex flex-col lg:flex-row lg:items-start lg:justify-between gap-3">
                      <div className="flex-1 min-w-0">
                        <h3 className="font-semibold text-lg text-gray-900 mb-1">{product.name}</h3>
                        <p className="text-gray-600 text-sm line-clamp-2 mb-3">
                          {product.description}
                        </p>
                        <div className="flex flex-wrap items-center gap-4">
                          <span className="text-black font-bold text-lg">
                            R$ {product.price ? parseFloat(product.price).toFixed(2) : '0,00'}
                          </span>
                          <span className="text-gray-500 text-sm capitalize">{product.category}</span>
                          
                          {/* Quantity Controls */}
                          <div className="flex items-center space-x-2">
                            <button
                              onClick={() => updateCartQuantity(item.product_id, item.quantity - 1)}
                              className="w-8 h-8 flex items-center justify-center border border-gray-300 rounded hover:bg-gray-100 transition-colors"
                            >
                              <FiMinus size={14} />
                            </button>
                            <span className="w-8 text-center font-semibold">{item.quantity}</span>
                            <button
                              onClick={() => updateCartQuantity(item.product_id, item.quantity + 1)}
                              className="w-8 h-8 flex items-center justify-center border border-gray-300 rounded hover:bg-gray-100 transition-colors"
                            >
                              <FiPlus size={14} />
                            </button>
                          </div>
                          
                          <span className="text-green-600 font-bold text-lg">
                            R$ {itemTotal.toFixed(2)}
                          </span>
                        </div>
                      </div>
                      
                      {/* Remove Button */}
                      <div className="flex flex-row sm:flex-col gap-2">
                        <button
                          onClick={() => removeFromCart(item.product_id)}
                          className="text-red-500 hover:text-red-700 p-2 rounded hover:bg-red-50 transition-colors"
                          title="Remover do carrinho"
                        >
                          <FiTrash2 size={18} />
                        </button>
                      </div>
                    </div>
                  </div>
                </div>
              </Card>
            );
          })}
        </div>

        {/* Cart Summary */}
        <Card className="p-6">
          <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 mb-6">
            <h3 className="text-xl font-semibold text-gray-900">Resumo do Pedido</h3>
            <span className="text-2xl font-bold text-green-600">
              R$ {cartTotal.toFixed(2)}
            </span>
          </div>
          
          <div className="space-y-3 mb-6">
            <div className="flex justify-between text-gray-600">
              <span>Subtotal</span>
              <span>R$ {cartTotal.toFixed(2)}</span>
            </div>
            <div className="flex justify-between text-gray-600">
              <span>Frete</span>
              <span>Grátis</span>
            </div>
            <div className="flex justify-between text-lg font-semibold text-gray-900 pt-3 border-t border-gray-200">
              <span>Total</span>
              <span>R$ {cartTotal.toFixed(2)}</span>
            </div>
          </div>

          <button
            onClick={processCheckout}
            className="w-full bg-green-600 hover:bg-green-700 text-white py-3 px-6 rounded-lg font-semibold text-lg transition-colors"
          >
            Finalizar Compra
          </button>
        </Card>
      </div>
    );
  };

  return (
    <div className="min-h-screen bg-gray-100 text-black" style={{ fontFamily: "Calibri, sans-serif" }}>
      {/* Mobile Navigation */}
      <MobileNavigation />

      <div className="flex">
        {/* Desktop Sidebar */}
        <DesktopSidebar />

        {/* Main Content */}
        <main className="flex-1 lg:ml-0 min-h-screen">
          {/* Mobile padding top */}
          <div className="lg:hidden pt-16"></div>
          
          {/* Tab Content */}
          <div className="p-4 lg:p-8">
            {/* Perfil */}
            {activeTab === "perfil" && (
              <ResponsiveContainer className="max-w-2xl">
                <Card className="p-6">
                  <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4 mb-6">
                    <h2 className="text-xl font-semibold text-gray-900">Informações do Usuário</h2>
                    <Button
                      variant={isEditingProfile ? "outline" : "primary"}
                      onClick={toggleEditProfile}
                    >
                      {isEditingProfile ? "Cancelar" : "Editar"}
                    </Button>
                  </div>
                  <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
                    <Input
                      label="Nome"
                      name="name"
                      value={user.name || ''}
                      onChange={handleProfileChange}
                      placeholder="Nome"
                      disabled={!isEditingProfile}
                    />
                    <Input
                      label="Email"
                      type="email"
                      name="email"
                      value={user.email || ''}
                      onChange={handleProfileChange}
                      placeholder="Email"
                      disabled={!isEditingProfile}
                    />
                    <Input
                      label="Telefone"
                      name="tel"
                      value={user.tel || ''}
                      onChange={handleProfileChange}
                      placeholder="Telefone"
                      disabled={!isEditingProfile}
                    />
                    <Input
                      label="CPF"
                      name="cpf"
                      value={user.cpf || ''}
                      onChange={handleProfileChange}
                      placeholder="CPF"
                      disabled={!isEditingProfile}
                    />
                    <Input
                      label="Data de Nascimento"
                      type="date"
                      name="birth_date"
                      value={user.birth_date || ''}
                      onChange={handleProfileChange}
                      disabled={!isEditingProfile}
                      className="md:col-span-2"
                    />
                  </div>
                  {isEditingProfile && (
                    <div className="flex justify-end pt-6">
                      <Button onClick={saveProfile}>
                        Salvar
                      </Button>
                    </div>
                  )}
                </Card>
              </ResponsiveContainer>
            )}

            {/* Endereços */}
            {activeTab === "enderecos" && (
              <ResponsiveContainer className="max-w-3xl">
                <Card className="p-6">
                  <h2 className="text-xl font-semibold mb-6 text-gray-900">Endereços</h2>
                  {user.addresses?.map((addr) => (
                    <div key={addr.id} className="border-b last:border-b-0 py-4 text-gray-900">
                      {editAddressId === addr.id ? (
                        <div className="grid grid-cols-1 md:grid-cols-2 gap-3 mb-4">
                          <Input
                            name="street"
                            value={addr.street || ''}
                            onChange={(e) => handleEditAddressChange(addr.id, e)}
                            placeholder="Rua"
                          />
                          <Input
                            name="number"
                            value={addr.number || ''}
                            onChange={(e) => handleEditAddressChange(addr.id, e)}
                            placeholder="Número"
                          />
                          <Input
                            name="city"
                            value={addr.city || ''}
                            onChange={(e) => handleEditAddressChange(addr.id, e)}
                            placeholder="Cidade"
                          />
                          <Input
                            name="state"
                            value={addr.state || ''}
                            onChange={(e) => handleEditAddressChange(addr.id, e)}
                            placeholder="Estado"
                          />
                          <Input
                            name="zip"
                            value={addr.zip || ''}
                            onChange={(e) => handleEditAddressChange(addr.id, e)}
                            placeholder="CEP"
                            className="md:col-span-2"
                          />
                          <button 
                            onClick={() => editAddress(addr.id)}
                            className="bg-green-600 text-white py-2 px-4 rounded col-span-full mt-2 hover:bg-green-700 transition-colors"
                          >
                            Salvar
                          </button>
                        </div>
                      ) : (
                        <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
                          <div>
                            <p className="font-medium text-gray-900">{addr.street}, {addr.number}</p>
                            <p className="text-gray-600">{addr.city} - {addr.state}, {addr.zip}</p>
                          </div>
                          <button 
                            onClick={() => setEditAddressId(addr.id)}
                            className="text-blue-600 hover:underline self-start sm:self-auto"
                          >
                            Editar
                          </button>
                        </div>
                      )}
                    </div>
                  ))}
                  <form className="mt-6 grid grid-cols-1 md:grid-cols-2 gap-4" onSubmit={saveAddress}>
                    <Input
                      name="street"
                      value={newAddress.street}
                      onChange={handleAddressChange}
                      placeholder="Rua"
                      required
                    />
                    <Input
                      name="number"
                      value={newAddress.number}
                      onChange={handleAddressChange}
                      placeholder="Número"
                      required
                    />
                    <Input
                      name="city"
                      value={newAddress.city}
                      onChange={handleAddressChange}
                      placeholder="Cidade"
                      required
                    />
                    <Input
                      name="state"
                      value={newAddress.state}
                      onChange={handleAddressChange}
                      placeholder="Estado"
                      required
                    />
                    <Input
                      name="zip"
                      value={newAddress.zip}
                      onChange={handleAddressChange}
                      placeholder="CEP"
                      required
                      className="md:col-span-2"
                    />
                    <button type="submit" className="bg-blue-600 text-white py-2 px-4 rounded col-span-full hover:bg-blue-700 transition-colors">
                      Adicionar
                    </button>
                  </form>
                </Card>
              </ResponsiveContainer>
            )}

            {/* Other tabs remain similar but wrapped with ResponsiveContainer and Card */}
            {activeTab === "desejos" && (
              <ResponsiveContainer className="max-w-6xl">
                <Card className="p-6">
                  <WishlistTab />
                </Card>
              </ResponsiveContainer>
            )}

            {activeTab === "carrinho" && (
              <ResponsiveContainer className="max-w-6xl">
                <Card className="p-6">
                  <CartTab />
                </Card>
              </ResponsiveContainer>
            )}

            {/* Add other tabs following the same pattern... */}
            {activeTab === "pedidos" && (
              <ResponsiveContainer className="max-w-3xl">
                <Card className="p-6">
                  <div className="text-center py-8">
                    <p className="text-gray-500">Histórico de pedidos</p>
                  </div>
                </Card>
              </ResponsiveContainer>
            )}

            {activeTab === "cupons" && (
              <ResponsiveContainer className="max-w-3xl">
                <Card className="p-6">
                  <div className="text-center py-8">
                    <p className="text-gray-500">Cupons</p>
                  </div>
                </Card>
              </ResponsiveContainer>
            )}

            {activeTab === "configuracoes" && (
              <ResponsiveContainer className="max-w-3xl">
                <Card className="p-6">
                  <div className="text-center py-8">
                    <p className="text-gray-500">Configurações</p>
                  </div>
                </Card>
              </ResponsiveContainer>
            )}

            {activeTab === "suporte" && (
              <ResponsiveContainer className="max-w-3xl">
                <Card className="p-6">
                  <div className="text-center py-8">
                    <p className="text-gray-500">Suporte</p>
                  </div>
                </Card>
              </ResponsiveContainer>
            )}
          </div>
        </main>
      </div>
    </div>
  );
};

export default ProfilePage;
